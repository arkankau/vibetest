import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

relu_cuda_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_bf16.h>

template <typename scalar_t, int VEC_SIZE>
__global__ void optimized_relu_kernel(const scalar_t* __restrict__ input,
                                      scalar_t* __restrict__ output,
                                      int64_t numel) {
    constexpr int ELEMENTS_PER_THREAD = 16 / sizeof(scalar_t);
    const int64_t tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int64_t stride = gridDim.x * blockDim.x;

    using vec_t = typename std::conditional<
        sizeof(scalar_t) == 4, float4,
        typename std::conditional<sizeof(scalar_t) == 2, uint4, void>::type
    >::type;

    for (int64_t base = tid * ELEMENTS_PER_THREAD; base < numel; base += stride * ELEMENTS_PER_THREAD) {
        vec_t vec_in;
        scalar_t elements[ELEMENTS_PER_THREAD];
        
        const scalar_t* addr = input + base;
        vec_in = *reinterpret_cast<const vec_t*>(addr);
        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            elements[v] = reinterpret_cast<const scalar_t*>(&vec_in)[v];
        }

        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            elements[v] = elements[v] > scalar_t(0) ? elements[v] : scalar_t(0);
        }

        vec_t vec_out;
        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            reinterpret_cast<scalar_t*>(&vec_out)[v] = elements[v];
        }

        scalar_t* out_addr = output + base;
        *reinterpret_cast<vec_t*>(out_addr) = vec_out;
    }

    // Handle remaining elements
    const int64_t remainder_start = (numel / ELEMENTS_PER_THREAD) * ELEMENTS_PER_THREAD;
    const int64_t linear_idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (linear_idx < numel - remainder_start) {
        const int64_t idx = remainder_start + linear_idx;
        output[idx] = max(input[idx], scalar_t(0));
    }
}

template <typename scalar_t>
__global__ void optimized_relu_kernel(const scalar_t* __restrict__ input,
                                      scalar_t* __restrict__ output,
                                      int64_t numel) {
    constexpr int ELEMENTS_PER_THREAD = 16 / sizeof(scalar_t);
    const int64_t tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int64_t stride = gridDim.x * blockDim.x;

    using vec_t = typename std::conditional<

        typename std::conditional<sizeof(scalar_t) == 2, uint4, void>::type

    for (int64_t base = tid * ELEMENTS_PER_THREAD; base < numel; base += stride * ELEMENTS_PER_THREAD) {
        vec_t vec_in;
        scalar_t elements[ELEMENTS_PER_THREAD];

        vec_in = *reinterpret_cast<const vec_t*>(addr);
        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            elements[v] = reinterpret_cast<const scalar_t*>(&vec_in)[v];
        }

        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            elements[v] = elements[v] > scalar_t(0) ? elements[v] : scalar_t(0);
        }

        vec_t vec_out;
        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            reinterpret_cast<scalar_t*>(&vec_out)[v] = elements[v];
        }

        scalar_t* out_addr = output + base;
        *reinterpret_cast<vec_t*>(out_addr) = vec_out;
    }

    // Handle remaining elements
    const int64_t remainder_start = (numel / ELEMENTS_PER_THREAD) * ELEMENTS_PER_THREAD;
    const int64_t linear_idx = blockIdx.x * blockDim.x + threadIdx.x;

}

#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_bf16.h>
#include <c10/cuda/CUDAStream.h>

template <typename scalar_t>
__global__ void optimized_relu_kernel(const scalar_t* __restrict__ input,
                                      scalar_t* __restrict__ output,
                                      int64_t numel) {
    constexpr int ELEMENTS_PER_THREAD = 16 / sizeof(scalar_t);
    const int64_t tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int64_t stride = gridDim.x * blockDim.x;

    using vec_t = typename std::conditional<
        sizeof(scalar_t) == 4, float4,
        typename std::conditional<sizeof(scalar_t) == 2, uint4,
        typename std::conditional<sizeof(scalar_t) == 8, double2, void>::type>::type
    >::type;

    static_assert(!std::is_same<vec_t, void>::value, "Unsupported scalar type");

    for (int64_t base = tid * ELEMENTS_PER_THREAD; base < numel; base += stride * ELEMENTS_PER_THREAD) {
        vec_t vec_in;
        scalar_t elements[ELEMENTS_PER_THREAD];
        
        const scalar_t* addr = input + base;
        vec_in = *reinterpret_cast<const vec_t*>(addr);
        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            elements[v] = reinterpret_cast<const scalar_t*>(&vec_in)[v];
        }

        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            elements[v] = elements[v] > scalar_t(0) ? elements[v] : scalar_t(0);
        }

        vec_t vec_out;
        #pragma unroll
        for (int v = 0; v < ELEMENTS_PER_THREAD; ++v) {
            reinterpret_cast<scalar_t*>(&vec_out)[v] = elements[v];
        }

        scalar_t* out_addr = output + base;
        *reinterpret_cast<vec_t*>(out_addr) = vec_out;
    }

    // Handle remaining elements
    const int64_t remainder_start = (numel / ELEMENTS_PER_THREAD) * ELEMENTS_PER_THREAD;
    const int64_t linear_idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (linear_idx < numel - remainder_start) {
        const int64_t idx = remainder_start + linear_idx;
        output[idx] = max(input[idx], scalar_t(0));
    }
}

torch::Tensor optimized_relu_cuda(torch::Tensor x) {
    auto output = torch::empty_like(x);
    const int64_t numel = x.numel();
    if (numel == 0) return output;

    constexpr int BLOCK_SIZE = 512;

    AT_DISPATCH_FLOATING_TYPES_AND2(
        at::ScalarType::Half, at::ScalarType::BFloat16,
        x.scalar_type(), "optimized_relu", [&] {
            constexpr int ELEMENTS_PER_THREAD = 16 / sizeof(scalar_t);
            const int GRID_SIZE = (numel + ELEMENTS_PER_THREAD * BLOCK_SIZE - 1) / (ELEMENTS_PER_THREAD * BLOCK_SIZE);
            
            optimized_relu_kernel<scalar_t><<<
                std::min(GRID_SIZE, 65535),
                BLOCK_SIZE,
                0,
                c10::cuda::getCurrentCUDAStream()
            >>>(
                x.data_ptr<scalar_t>(),
                output.data_ptr<scalar_t>(),
                numel
            );
        });
    
    return output;
}

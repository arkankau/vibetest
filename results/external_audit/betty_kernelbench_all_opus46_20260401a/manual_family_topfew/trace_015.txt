__global__ void optimized_tanh_kernel(const float* input, float* output, int num_elements) {
    const int stride = blockDim.x * gridDim.x * 4;
    int idx = (blockIdx.x * blockDim.x + threadIdx.x) * 4;
    
    float4 in, out;
    for(; idx < num_elements; idx += stride) {
        in = *reinterpret_cast<const float4*>(input + idx);

    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int idx = tid * 4;
    
    if (idx >= num_elements) return;
    
    float4 in = *reinterpret_cast<const float4*>(input + idx);
    float4 out;
    out.x = tanhf(in.x);
    out.y = tanhf(in.y);
    out.z = tanhf(in.z);

    *reinterpret_cast<float4*>(output + idx) = out;
}

torch::Tensor fast_tanh_cuda(torch::Tensor input) {
    torch::Tensor output = torch::empty_like(input);
    int num_elements = input.numel();
    
    const int threads_per_block = 1024;
    const int elements_per_thread = 4;
    int num_blocks = (num_elements / elements_per_thread + threads_per_block - 1) / threads_per_block;
    
    optimized_tanh_kernel<<<num_blocks, threads_per_block>>>(
        input.data_ptr<float>(),
        output.data_ptr<float>(),

}

    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int stride = gridDim.x * blockDim.x * elements_per_thread;

) {
    int b = blockIdx.x;

    int spatial_size = H * W;
    int elements_per_thread = (spatial_size + blockDim.x - 1) / blockDim.x;
    int start = threadIdx.x * elements_per_thread;
    int end = min(start + elements_per_thread, spatial_size);
    
    extern __shared__ float smem[];
    float* sum_smem = smem;
    float* sum_sq_smem = &smem[blockDim.x];
    
    float sum = 0.0f;
    float sum_sq = 0.0f;
    
    for (int i = start; i < end; ++i) {
        int h = i / W;
        int w = i % W;
        int idx = ((b * num_features + f) * H + h) * W + w;
        float val = input[idx];
        sum += val;
        sum_sq += val * val;
    }
    
    sum_smem[threadIdx.x] = sum;
    sum_sq_smem[threadIdx.x] = sum_sq;
    __syncthreads();
    
    for (int stride = blockDim.x / 2; stride > 0; stride >>= 1) {
        if (threadIdx.x < stride) {
            sum_smem[threadIdx.x] += sum_smem[threadIdx.x + stride];
            sum_sq_smem[threadIdx.x] += sum_sq_smem[threadIdx.x + stride];
        }
        __syncthreads();
    }
    
    float total_sum = sum_smem[0];
    float total_sum_sq = sum_sq_smem[0];
    
    float mean = total_sum / spatial_size;
    float variance = (total_sum_sq / spatial_size) - (mean * mean) + epsilon;
    float inv_std = rsqrtf(variance);

        int w = i % W;
        int idx = ((b * num_features + f) * H + h) * W + w;

template <typename T>
__inline__ __device__ T warpReduceSum(T val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset /= 2) 
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__global__ void instance_norm_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    const int batch_size,
    const int num_features,
    const int H,
    const int W,
    const float epsilon
) {
    const int b = blockIdx.x;
    const int f = blockIdx.y;
    const int spatial_size = H * W;
    const float inv_spatial_size = 1.0f / spatial_size;
    
    if (b >= batch_size || f >= num_features) return;

    const int base_idx = (b * num_features + f) * spatial_size;
    const int elements_per_block = (spatial_size + blockDim.x * 4 - 1) / (blockDim.x * 4);
    const int start = threadIdx.x * elements_per_block * 4;
    const int end = min(start + elements_per_block * 4, spatial_size);

    // Vectorized load and compute
    for (int i = start; i < end; i += 4) {
        float4 val = *reinterpret_cast<const float4*>(input + base_idx + i);
        sum += val.x + val.y + val.z + val.w;
        sum_sq += val.x * val.x + val.y * val.y + val.z * val.z + val.w * val.w;
    }

    // Warp-level reduction
    sum = warpReduceSum(sum);
    sum_sq = warpReduceSum(sum_sq);

    // First thread in warp stores partial sums
    __shared__ float smem_sum[32], smem_sq[32];
    if (threadIdx.x % 32 == 0) {
        smem_sum[threadIdx.x/32] = sum;
        smem_sq[threadIdx.x/32] = sum_sq;

        sum = threadIdx.x < blockDim.x/32 ? smem_sum[threadIdx.x] : 0;
        sum_sq = threadIdx.x < blockDim.x/32 ? smem_sq[threadIdx.x] : 0;

        sum_sq = warpReduceSum(sum_sq);

            smem_sq[0] = sum_sq;

__global__ void instance_norm_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    const int batch_size,
    const int num_features,
    const int H,
    const int W,
    const float epsilon
) {
    const int b = blockIdx.x;
    const int f = blockIdx.y;
    const int spatial_size = H * W;
    const float inv_spatial_size = 1.0f / spatial_size;
    
    if (b >= batch_size || f >= num_features) return;

    const int base_idx = (b * num_features + f) * spatial_size;
    const int elements_per_thread = (spatial_size + blockDim.x*4 - 1) / (blockDim.x*4);
    const int start = threadIdx.x * elements_per_thread * 4;
    const int end = min(start + elements_per_thread*4, spatial_size);

    float sum = 0.0f, sum_sq = 0.0f;

    // Vectorized load with edge handling
    for (int i = start; i < end; i += 4) {
        float4 val;
        if (i + 3 < spatial_size) {
            val = __ldg(reinterpret_cast<const float4*>(input + base_idx + i));
        } else {
            val.x = (i < spatial_size) ? input[base_idx + i] : 0.0f;
            val.y = (i+1 < spatial_size) ? input[base_idx + i+1] : 0.0f;
            val.z = (i+2 < spatial_size) ? input[base_idx + i+2] : 0.0f;
            val.w = (i+3 < spatial_size) ? input[base_idx + i+3] : 0.0f;
        }
        sum += val.x + val.y + val.z + val.w;
        sum_sq += val.x*val.x + val.y*val.y + val.z*val.z + val.w*val.w;
    }

    // Warp reduction
    sum = warpReduceSum(sum);
    sum_sq = warpReduceSum(sum_sq);

    // Block reduction
    __shared__ float smem_sum[8], smem_sq[8];
    const int wid = threadIdx.x / 32;
    const int lane = threadIdx.x % 32;

    if (lane == 0) {
        smem_sum[wid] = sum;
        smem_sq[wid] = sum_sq;
    }
    __syncthreads();

    if (wid == 0) {
        sum = lane < blockDim.x/32 ? smem_sum[lane] : 0.0f;
        sum_sq = lane < blockDim.x/32 ? smem_sq[lane] : 0.0f;
        sum = warpReduceSum(sum);
        sum_sq = warpReduceSum(sum_sq);
    }
    __syncthreads();

    const float mean = sum * inv_spatial_size;
    const float inv_std = rsqrtf(sum_sq * inv_spatial_size - mean * mean + epsilon);

    // Vectorized write with edge handling
    for (int i = start; i < end; i += 4) {
        float4 val;
        if (i + 3 < spatial_size) {
            val = __ldg(reinterpret_cast<const float4*>(input + base_idx + i));
        } else {

            val.y = (i+1 < spatial_size) ? input[base_idx + i+1] : 0.0f;
            val.z = (i+2 < spatial_size) ? input[base_idx + i+2] : 0.0f;
            val.w = (i+3 < spatial_size) ? input[base_idx + i+3] : 0.0f;

    #pragma unroll

        val += __shfl_xor_sync(0xffffffff, val, mask);

__global__ void instance_norm_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    const int batch_size,
    const int num_features,
    const int H,
    const int W,
    const float epsilon
) {
    extern __shared__ float smem[];
    const int num_warps = blockDim.x / 32;
    float* smem_sum = smem;
    float* smem_sq = smem + num_warps;
    
    const int b = blockIdx.x;
    const int f = blockIdx.y;
    const int spatial_size = H * W;
    const float inv_spatial_size = 1.0f / spatial_size;
    
    if (b >= batch_size || f >= num_features) return;

    const int base_idx = (b * num_features + f) * spatial_size;
    const int num_float4 = spatial_size / 4;
    const int elements_per_thread = (num_float4 + blockDim.x - 1) / blockDim.x;
    const int start = threadIdx.x * elements_per_thread;
    const int end = min(start + elements_per_thread, num_float4);

        sum_sq += vec.x*vec.x + vec.y*vec.y + vec.z*vec.z + vec.w*vec.w;
    }

    // Warp-level reduction
    sum = warpReduceSum(sum);
    sum_sq = warpReduceSum(sum_sq);

    // Block-level reduction
    const int wid = threadIdx.x / 32;
    const int lane = threadIdx.x % 32;

        smem_sum[wid] = sum;
        smem_sq[wid] = sum_sq;

        sum_sq = lane < num_warps ? smem_sq[lane] : 0.0f;
        sum = warpReduceSum(sum);
        sum_sq = warpReduceSum(sum_sq);

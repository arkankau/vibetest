import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

hinge_kernel = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void hinge_forward(
    const float* __restrict__ pred,
    const float* __restrict__ targ,
    float* output,
    int N
) {
    const int TILE = 4;
    __shared__ float smem[256];
    
    float sum = 0.0f;
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    
    // Vectorized load for 4 elements
    for(int i = tid*TILE; i < N; i += blockDim.x*gridDim.x*TILE) {
        float4 p = reinterpret_cast<const float4*>(pred)[i/TILE];
        float4 t = reinterpret_cast<const float4*>(targ)[i/TILE];
        
        sum += fmaxf(1.0f - p.x*t.x, 0.0f);
        sum += fmaxf(1.0f - p.y*t.y, 0.0f);
        sum += fmaxf(1.0f - p.z*t.z, 0.0f);
        sum += fmaxf(1.0f - p.w*t.w, 0.0f);
    }
    
    // Scalar remainder
    for(int i = tid + blockDim.x*gridDim.x*(N/(blockDim.x*gridDim.x*TILE))*TILE; i < N; ++i) {
        float val = 1.0f - pred[i] * targ[i];
        sum += fmaxf(val, 0.0f);
    }
    
    // Warp reduce
    for(int offset = 16; offset > 0; offset >>= 1)
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    
    if(threadIdx.x % 32 == 0)
        smem[threadIdx.x/32] = sum;
    __syncthreads();
    
    // Block reduce
    if(threadIdx.x < 32) {
        sum = threadIdx.x < (blockDim.x+31)/32 ? smem[threadIdx.x] : 0;
        for(int offset = 16; offset > 0; offset >>= 1)

}

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

hinge_kernel = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void hinge_forward(
    const float* __restrict__ pred,
    const float* __restrict__ targ,
    float* output,
    int N
) {
    const int VEC = 4;
    extern __shared__ float smem[];
    
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    float sum = 0.0f;
    
    // Vectorized main loop
    int num_vec = N / VEC;
    int stride = gridDim.x * blockDim.x;
    for(int i = tid; i < num_vec; i += stride) {
        float4 p = reinterpret_cast<const float4*>(pred)[i];
        float4 t = reinterpret_cast<const float4*>(targ)[i];
        sum += fmaxf(1.0f - p.x*t.x, 0.0f);
        sum += fmaxf(1.0f - p.y*t.y, 0.0f);
        sum += fmaxf(1.0f - p.z*t.z, 0.0f);
        sum += fmaxf(1.0f - p.w*t.w, 0.0f);
    }
    
    // Scalar remainder
    int rem_start = num_vec * VEC;
    for(int i = rem_start + tid; i < N; i += stride) {
        sum += fmaxf(1.0f - pred[i]*targ[i], 0.0f);
    }
    
    // Warp reduction
    for(int offset = 16; offset > 0; offset >>= 1)
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    
    if(threadIdx.x % 32 == 0)
        smem[threadIdx.x/32] = sum;
    __syncthreads();
    
    // Block reduction
    if(threadIdx.x < 32) {
        float warp_sum = threadIdx.x < (blockDim.x+31)/32 ? smem[threadIdx.x] : 0;
        for(int offset = 16; offset > 0; offset >>= 1)
            warp_sum += __shfl_down_sync(0xffffffff, warp_sum, offset);
        if(threadIdx.x == 0)
            atomicAdd(output, warp_sum);
    }
}

torch::Tensor hinge_cuda(torch::Tensor pred, torch::Tensor targ) {
    auto output = torch::zeros({}, pred.options());
    int N = pred.numel();
    
    const int block_size = 512;
    int grid_size = min((N + block_size*4 - 1) / (block_size*4), 512);

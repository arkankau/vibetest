// Optimized mean reduction with warp-level parallelism
__global__ void mean_channel_kernel(const float* input, float* output, int N, int C, int D, int H, int W) {
    const int tid = threadIdx.x;
    const int spatial_idx = blockIdx.x * blockDim.x + tid;
    const int spatial_size = D * H * W;
    
    if (spatial_idx >= spatial_size) return;
    
    for(int n = 0; n < N; ++n) {
        float sum = 0.0f;
        const float* channel_ptr = input + (n * C * spatial_size) + spatial_idx;

        #pragma unroll
        for(int c = 0; c < 16; ++c) {
            sum += channel_ptr[c * spatial_size];

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

fused_ops_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>

__global__ void unified_ops_kernel(
    const float* __restrict__ conv_out,
    const float* __restrict__ bias,
    float* __restrict__ final_out,
    float scaling_factor,
    int N, int C, int D, int H, int W
) {
    const int spatial_size = D * H * W;
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    if (tid >= N * spatial_size) return;
    
    const int n = tid / spatial_size;
    const int s_idx = tid % spatial_size;
    const int base_offset = n * C * spatial_size + s_idx;

    float sum = 0.0f;
    #pragma unroll
    for (int c = 0; c < 16; ++c) {
        sum += __ldg(&conv_out[base_offset + c * spatial_size]);
    }

    const float mean_bias = sum * 0.0625f + __ldg(bias);
    final_out[tid] = __fmul_rn(tanhf(mean_bias), scaling_factor);

__global__ void fused_postprocess_kernel(
    const float* conv_out,
    float* output,
    float bias,
    float scaling_factor,
    int C,
    int elements_per_sample,
    int total_elements
) {
    const int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total_elements) return;

    const int n = idx / elements_per_sample;
    const int spatial_idx = idx % elements_per_sample;

        sum += conv_out[(n * C + c) * elements_per_sample + spatial_idx];

    output[n * elements_per_sample + spatial_idx] = tanhf(sum) * scaling_factor;
}

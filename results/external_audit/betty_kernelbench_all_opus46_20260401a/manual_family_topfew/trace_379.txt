#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__device__ float gelu(float x) {
    return 0.5f * x * (1.0f + erff(x / 1.41421356237f));
}

__global__ void fused_gemm_ops_kernel(
    const float* input,
    const float* weight,
    const float* bias,
    float* output,
    int batch_size,
    int in_features,
    int out_features,
    int max_dim
) {
    // 2D grid: x-dim for batch, y-dim for output features
    int bid = blockIdx.x;
    int tid = threadIdx.x;
    
    extern __shared__ float shared_max[];
    
    // GEMM computation
    float val = 0.0f;
    for(int i = tid; i < in_features; i += blockDim.x) {
        val += input[bid * in_features + i] * weight[blockIdx.y * in_features + i];
    }
    val = warpReduceSum(val);
    
    if(tid == 0) {
        val += bias[blockIdx.y];
        __shared__ float row_max;
        
        // Store GEMM result and find max
        if(max_dim == 1) {
            output[bid * out_features + blockIdx.y] = val;
            if(blockIdx.y == 0) row_max = -INFINITY;
            __syncthreads();
            atomicMaxFloat(&row_max, val);
            __syncthreads();
            
            // Subtract mean and apply GELU
            float result = gelu(val - row_max);
            output[bid * out_features + blockIdx.y] = result;
        }
    }
}

// Helper functions for warp reduction
__inline__ __device__ float warpReduceSum(float val) {
    for(int offset = 16; offset > 0; offset /= 2)
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__device__ void atomicMaxFloat(float* address, float val) {
    int* address_as_int = (int*)address;
    int old = *address_as_int, assumed;
    do {
        assumed = old;
        if(__int_as_float(assumed) < val)
            old = atomicCAS(address_as_int, assumed, __float_as_int(val));
    } while(assumed != old);
}

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__device__ float gelu(float x) {
    return 0.5f * x * (1.0f + erff(x / 1.41421356237f));
}

__inline__ __device__ float warpReduceSum(float val) {
    for(int offset = 16; offset > 0; offset /= 2)
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__device__ void atomicMaxFloat(float* address, float val) {
    int* address_as_int = (int*)address;
    int old = *address_as_int, assumed;
    do {
        assumed = old;
        if(__int_as_float(assumed) < val)
            old = atomicCAS(address_as_int, assumed, __float_as_int(val));
    } while(assumed != old);
}

__global__ void fused_gemm_ops_kernel(
    const float* input,
    const float* weight,
    const float* bias,
    float* output,
    int batch_size,
    int in_features,
    int out_features,
    int max_dim
) {
    int bid = blockIdx.x;
    int tid = threadIdx.x;
    
    extern __shared__ float shared_max[];
    
    // GEMM computation
    float val = 0.0f;
    for(int i = tid; i < in_features; i += blockDim.x) {
        val += input[bid * in_features + i] * weight[blockIdx.y * in_features + i];
    }
    val = warpReduceSum(val);
    
    if(tid == 0) {
        val += bias[blockIdx.y];
        shared_max[blockIdx.y] = val;  // Store GEMM result
        
        // Find max across columns
        if(max_dim == 1) {
            __shared__ float row_max;
            if(threadIdx.x == 0) {
                row_max = -INFINITY;
                for(int i = 0; i < out_features; i++) {
                    if(shared_max[i] > row_max) row_max = shared_max[i];
                }
            }
            __syncthreads();
            
            // Subtract mean and apply GELU
            float result = gelu(val - row_max);
            output[bid * out_features + blockIdx.y] = result;
        }
    }
}

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

fused_ops_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

#define BLOCK_SIZE 256
#define UNROLL_FACTOR 4

__device__ float fast_gelu(float x) {
    const float sqrt_2_over_pi = 0.7978845608028654f;
    const float coef = 0.044715f;
    float x_cubed = x * x * x;
    float tanh_arg = sqrt_2_over_pi * (x + coef * x_cubed);
    return 0.5f * x * (1.0f + tanhf(tanh_arg));
}

__global__ void fused_gemm_ops_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int in_features,
    int out_features
) {
    extern __shared__ float shared_input[];
    int batch_idx = blockIdx.x;
    int tid = threadIdx.x;

    // Load input into shared memory with coalesced access
    #pragma unroll
    for(int i = tid; i < in_features; i += blockDim.x) {
        shared_input[i] = input[batch_idx * in_features + i];
    }
    __syncthreads();

    float sum = 0.0f;
    float max_val = -INFINITY;
    
    // Process multiple features per thread with loop unrolling
    for(int f_base = 0; f_base < out_features; f_base += BLOCK_SIZE * UNROLL_FACTOR) {
        float vals[UNROLL_FACTOR] = {0};
        
        #pragma unroll
        for(int i = 0; i < in_features; ++i) {
            float w[UNROLL_FACTOR];
            int f_idx = f_base + tid * UNROLL_FACTOR;
            #pragma unroll
            for(int u = 0; u < UNROLL_FACTOR; ++u) {
                if(f_idx + u < out_features) {
                    w[u] = weight[i * out_features + f_idx + u];
                }
            }
            #pragma unroll
            for(int u = 0; u < UNROLL_FACTOR; ++u) {
                vals[u] += shared_input[i] * w[u];
            }
        }

        #pragma unroll
        for(int u = 0; u < UNROLL_FACTOR; ++u) {
            int f_idx = f_base + tid * UNROLL_FACTOR + u;
            if(f_idx < out_features) {
                float val = vals[u] + bias[f_idx];
                sum += val;
                max_val = fmaxf(max_val, val);
            }
        }
    }

    // Warp-level reduction for sum and max
    float warp_sum = sum;
    float warp_max = max_val;
    for(int offset = 16; offset > 0; offset >>= 1) {
        warp_sum += __shfl_down_sync(0xffffffff, warp_sum, offset);
        warp_max = fmaxf(warp_max, __shfl_down_sync(0xffffffff, warp_max, offset));
    }

    if((tid & 31) == 0) {
        atomicAdd(&shared_input[tid >> 5], warp_sum);
        atomicMax((int*)&shared_input[(tid >> 5) + 32], __float_as_int(warp_max));
    }
    __syncthreads();

    if(tid == 0) {
        float total_sum = 0, total_max = -INFINITY;
        for(int i = 0; i < 8; ++i) {
            total_sum += shared_input[i];
            total_max = fmaxf(total_max, __int_as_float((int)shared_input[i + 32]));
        }
        output[batch_idx] = fast_gelu(total_max - (total_sum / out_features));
    }
}

__device__ float fast_gelu(float x) {
    const float sqrt_2_over_pi = 0.7978845608028654f;
    const float coef = 0.044715f;
    float x_cubed = x * x * x;
    float tanh_arg = sqrt_2_over_pi * (x + coef * x_cubed);
    return 0.5f * x * (1.0f + tanhf(tanh_arg));
}

__global__ void fused_gemm_ops_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int in_features,
    int out_features
) {
    extern __shared__ float shared_input[];
    int batch_idx = blockIdx.x;
    int tid = threadIdx.x;

    // Load input into shared memory with coalesced access
    #pragma unroll
    for(int i = tid; i < in_features; i += blockDim.x) {
        shared_input[i] = input[batch_idx * in_features + i];
    }
    __syncthreads();

    float sum = 0.0f;
    float max_val = -INFINITY;

    for(int f_base = 0; f_base < out_features; f_base += BLOCK_SIZE * UNROLL_FACTOR) {
        float vals[UNROLL_FACTOR] = {0};
        
        #pragma unroll
        for(int i = 0; i < in_features; ++i) {
            float w[UNROLL_FACTOR];
            int f_idx = f_base + tid * UNROLL_FACTOR;
            #pragma unroll
            for(int u = 0; u < UNROLL_FACTOR; ++u) {
                if(f_idx + u < out_features) {
                    w[u] = weight[i * out_features + f_idx + u];
                }
            }
            #pragma unroll
            for(int u = 0; u < UNROLL_FACTOR; ++u) {
                vals[u] += shared_input[i] * w[u];
            }
        }

        for(int u = 0; u < UNROLL_FACTOR; ++u) {

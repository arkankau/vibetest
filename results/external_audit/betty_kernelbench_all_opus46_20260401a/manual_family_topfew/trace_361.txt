import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

lse_activations_kernel = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__device__ __forceinline__ float gelu(float x) {
    const float kBeta = 0.7978845608028654f;  // sqrt(2/pi)
    const float kKappa = 0.044715f;
    float x_cubed = x * x * x;
    float tanh_arg = kBeta * (x + kKappa * x_cubed);
    return 0.5f * x * (1.0f + tanhf(tanh_arg));
}

__global__ void lse_activations_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    int num_features,
    int num_elements
) {
    extern __shared__ float smem[];
    int batch_idx = blockIdx.x;
    int tid = threadIdx.x;

    // Load data into shared memory
    float val = tid < num_features ? input[batch_idx * num_features + tid] : -FLT_MAX;
    smem[tid] = val;
    __syncthreads();

    // Find max for numerical stability
    for(int s = blockDim.x/2; s > 0; s >>= 1) {
        if(tid < s) smem[tid] = fmaxf(smem[tid], smem[tid + s]);
        __syncthreads();
    }
    float max_val = smem[0];
    __syncthreads();

    // Compute exp(x - max)
    val = tid < num_features ? expf(val - max_val) : 0.0f;
    smem[tid] = val;
    __syncthreads();

    // Sum exponents
    for(int s = blockDim.x/2; s > 0; s >>= 1) {
        if(tid < s) smem[tid] += smem[tid + s];
        __syncthreads();
    }
    float log_sum = logf(smem[0]) + max_val;

    // Apply activations (only first thread writes)
    if(tid == 0) {
        // Combined LeakyReLU (0.01^2 = 0.0001)
        float x = log_sum > 0 ? log_sum : 0.0001f * log_sum;
        
        // Apply GELU twice
        x = gelu(gelu(x));
        output[batch_idx] = x;
    }
}

torch::Tensor fused_op_cuda(torch::Tensor input) {
    auto output = torch::empty({input.size(0), 1}, input.options());
    
    dim3 grid(input.size(0));
    dim3 block((input.size(1) + 31) / 32 * 32);  // Round up to nearest 32
    size_t smem = input.size(1) * sizeof(float);
    
    lse_activations_kernel<<<grid, block, smem>>>(
        input.data_ptr<float>(),
        output.data_ptr<float>(),
        input.size(1),
        input.numel()
    );

}
"""

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

lse_activations_kernel = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__device__ __forceinline__ float gelu(float x) {
    return 0.5f * x * (1.0f + tanhf(0.7978845608028654f * (x + 0.044715f * x * x * x)));
}

__global__ void lse_activations_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    int num_features,
    int num_elements
) {
    extern __shared__ float smem[];
    const int batch_idx = blockIdx.x;
    const int tid = threadIdx.x;
    const int warp_id = tid >> 5;  // tid / 32
    const int lane_id = tid & 0x1f;  // tid % 32

    // Load input with coalesced access
    float val = tid < num_features ? input[batch_idx * num_features + tid] : -FLT_MAX;

    // Warp-wise max reduction
    float warp_max = val;
    #pragma unroll
    for(int offset=16; offset>0; offset>>=1)
        warp_max = fmaxf(warp_max, __shfl_down_sync(0xffffffff, warp_max, offset));

    // Store warp maxima to shared memory
    if(lane_id == 0) smem[warp_id] = warp_max;
    __syncthreads();

    // Global max reduction (first warp only)
    float global_max = -FLT_MAX;
    if(warp_id == 0) {
        float v = lane_id < (blockDim.x>>5) ? smem[lane_id] : -FLT_MAX;
        #pragma unroll
        for(int offset=16; offset>0; offset>>=1)
            v = fmaxf(v, __shfl_down_sync(0xffffffff, v, offset));
        global_max = v;
    }
    global_max = __shfl_sync(0xffffffff, global_max, 0);

    // Compute exp with numerical stability
    float exp_val = tid < num_features ? expf(val - global_max) : 0.0f;

    // Warp-wise sum reduction
    float warp_sum = exp_val;
    #pragma unroll
    for(int offset=16; offset>0; offset>>=1)
        warp_sum += __shfl_down_sync(0xffffffff, warp_sum, offset);

    // Store warp sums to shared memory
    if(lane_id == 0) smem[warp_id] = warp_sum;
    __syncthreads();

    // Global sum reduction (first warp only)
    float global_sum = 0.0f;
    if(warp_id == 0) {
        float v = lane_id < (blockDim.x>>5) ? smem[lane_id] : 0.0f;
        #pragma unroll
        for(int offset=16; offset>0; offset>>=1)
            v += __shfl_down_sync(0xffffffff, v, offset);
        global_sum = v;
    }
    global_sum = __shfl_sync(0xffffffff, global_sum, 0);

    // Final computations and activations
    if(tid == 0) {
        float log_sum = logf(global_sum) + global_max;
        
        // Dual LeakyReLU
        log_sum = log_sum > 0 ? log_sum : 0.01f * log_sum;
        log_sum = log_sum > 0 ? log_sum : 0.01f * log_sum;
        
        // Dual GELU with fast approximation
        log_sum = gelu(gelu(log_sum));
        
        output[batch_idx] = log_sum;
    }
}

torch::Tensor fused_op_cuda(torch::Tensor input) {
    const int num_features = input.size(1);
    int block_size = 1;
    while(block_size < num_features) block_size <<= 1;
    block_size = min(block_size, 1024);
    
    auto output = torch::empty({input.size(0), 1}, input.options());
    
    dim3 grid(input.size(0));
    dim3 block(block_size);
    size_t smem = (block_size / 32) * sizeof(float);
    
    lse_activations_kernel<<<grid, block, smem>>>(
        input.data_ptr<float>(),
        output.data_ptr<float>(),
        num_features,
        input.numel()
    );
    
    return output;
}
"""

__global__ void lse_activations_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    int num_features,
    int num_elements
) {
    extern __shared__ float smem[];
    const int batch_idx = blockIdx.x;
    const int tid = threadIdx.x;
    const int warp_id = tid >> 5;
    const int lane_id = tid & 0x1f;

    float val = tid < num_features ? input[batch_idx * num_features + tid] : -FLT_MAX;

    float warp_max = val;
    #pragma unroll
    for(int offset=16; offset>0; offset>>=1)
        warp_max = fmaxf(warp_max, __shfl_down_sync(0xffffffff, warp_max, offset));

    if(lane_id == 0) smem[warp_id] = warp_max;
    __syncthreads();

    float global_max = -FLT_MAX;
    if(warp_id == 0) {
        float v = lane_id < (blockDim.x>>5) ? smem[lane_id] : -FLT_MAX;
        #pragma unroll
        for(int offset=16; offset>0; offset>>=1)
            v = fmaxf(v, __shfl_down_sync(0xffffffff, v, offset));
        global_max = v;
    }
    global_max = __shfl_sync(0xffffffff, global_max, 0);

    float warp_sum = exp_val;
    #pragma unroll

        warp_sum += __shfl_down_sync(0xffffffff, warp_sum, offset);

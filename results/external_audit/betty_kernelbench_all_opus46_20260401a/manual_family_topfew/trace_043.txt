import torch

mean_kernel_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

const int WARP_SIZE = 32;

__device__ __forceinline__ float warp_reduce(float sum) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) 
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    return sum;
}

__global__ void optimized_mean_kernel(const float* __restrict__ input, float* __restrict__ output,
                                     int num_batches, int inner_size, int reduce_size) {
    // Transposed layout: [B, inner, reduce]
    const int batch = blockIdx.z;
    const int inner = blockIdx.x * blockDim.x + threadIdx.x;
    const int stride = gridDim.x * blockDim.x;

    if (batch >= num_batches || inner >= inner_size) return;

    float sum = 0.0f;
    for (int r = threadIdx.y; r < reduce_size; r += blockDim.y) {
        sum += input[((batch * inner_size + inner) * reduce_size) + r];
    }

    // Warp reduction
    sum = warp_reduce(sum);
    
    __shared__ float smem[WARP_SIZE];
    if (threadIdx.y == 0) {
        smem[threadIdx.x] = 0.0f;
    }
    __syncthreads();

    if (threadIdx.y % WARP_SIZE == 0) {
        atomicAdd(&smem[threadIdx.x], sum);
    }
    __syncthreads();

    if (threadIdx.y == 0) {
        sum = smem[threadIdx.x] / reduce_size;
        output[batch * inner_size + inner] = sum;
    }
}

torch::Tensor fused_mean_cuda(torch::Tensor input, int dim) {
    TORCH_CHECK(dim == 2, "Optimized for transposed dim=2 reduction");

        input.contiguous().data_ptr<float>(),
        output.data_ptr<float>(),
        num_batches,
        inner_size,
        reduce_size
    );

}

__global__ void optimized_mean_kernel(const float* __restrict__ input, float* __restrict__ output,

    extern __shared__ float smem[];

    const int inner = blockIdx.x * blockDim.x + threadIdx.x;
    const int tid = threadIdx.y;
    const int lane_id = threadIdx.x;

    if (inner >= inner_size || batch >= num_batches) return;

    const int chunk_size = 4;
    const int chunks = (reduce_size + (blockDim.y*chunk_size) - 1) / (blockDim.y*chunk_size);
    
    float sum = 0.0f;
    #pragma unroll
    for (int chunk = 0; chunk < chunks; ++chunk) {
        int offset = chunk * blockDim.y * chunk_size + tid * chunk_size;
        #pragma unroll
        for (int i = 0; i < chunk_size; ++i) {
            int r = offset + i;
            if (r < reduce_size) {

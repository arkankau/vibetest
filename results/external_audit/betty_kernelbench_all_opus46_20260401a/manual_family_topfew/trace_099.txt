__global__ void triplet_margin_loss_kernel(const float* anchor, const float* positive, const float* negative,
                                           float* output_loss, float margin, int batch_size, int feature_size) {
    extern __shared__ float shared[];
    int bid = blockIdx.x;
    int tid = threadIdx.x;
    int idx = bid * feature_size;

    float sum_ap = 0.0f, sum_an = 0.0f;

        float a = anchor[idx + i];
        float p = positive[idx + i];
        float n = negative[idx + i];

        sum_an += (a - n) * (a - n);

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

triplet_margin_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

const int VEC_SIZE = 4;
const int BLOCK_SIZE = 512;

__inline__ __device__ float warp_reduce(float val) {
    val += __shfl_down_sync(0xffffffff, val, 16);
    val += __shfl_down_sync(0xffffffff, val, 8);
    val += __shfl_down_sync(0xffffffff, val, 4);
    val += __shfl_down_sync(0xffffffff, val, 2);
    return val + __shfl_down_sync(0xffffffff, val, 1);
}

__inline__ __device__ float block_reduce(float val, float* shared) {
    int lane = threadIdx.x % 32;
    int wid = threadIdx.x / 32;
    
    val = warp_reduce(val);
    if (lane == 0) shared[wid] = val;
    __syncthreads();
    
    val = (threadIdx.x < blockDim.x / 32) ? shared[lane] : 0.0f;
    return warp_reduce(val);
}

__global__ void triplet_margin_kernel(
    const float* __restrict__ anchor,
    const float* __restrict__ positive,
    const float* __restrict__ negative,
    float* output_loss,
    float margin,
    int batch_size,
    int feature_size
) {
    extern __shared__ float smem[];
    
    const int vec_feature_size = feature_size / VEC_SIZE;
    const int vec_idx = blockIdx.x * vec_feature_size;
    
    float4 a_vec, p_vec, n_vec;
    float sum_ap = 0.0f, sum_an = 0.0f;
    
    for (int i = threadIdx.x; i < vec_feature_size; i += BLOCK_SIZE) {
        a_vec = *reinterpret_cast<const float4*>(&anchor[vec_idx + i*VEC_SIZE]);
        p_vec = *reinterpret_cast<const float4*>(&positive[vec_idx + i*VEC_SIZE]);
        n_vec = *reinterpret_cast<const float4*>(&negative[vec_idx + i*VEC_SIZE]);
        
        sum_ap += (a_vec.x - p_vec.x)*(a_vec.x - p_vec.x) + 
                 (a_vec.y - p_vec.y)*(a_vec.y - p_vec.y) +
                 (a_vec.z - p_vec.z)*(a_vec.z - p_vec.z) +
                 (a_vec.w - p_vec.w)*(a_vec.w - p_vec.w);
                 
        sum_an += (a_vec.x - n_vec.x)*(a_vec.x - n_vec.x) + 
                 (a_vec.y - n_vec.y)*(a_vec.y - n_vec.y) +
                 (a_vec.z - n_vec.z)*(a_vec.z - n_vec.z) +
                 (a_vec.w - n_vec.w)*(a_vec.w - n_vec.w);
    }
    
    sum_ap = block_reduce(sum_ap, smem);
    sum_an = block_reduce(sum_an, smem);
    
    if (threadIdx.x == 0) {
        float d_ap = sqrtf(sum_ap + 1e-6f);
        float d_an = sqrtf(sum_an + 1e-6f);
        float loss = fmaxf(d_ap - d_an + margin, 0.0f);
        atomicAdd(output_loss, loss / batch_size);
    }
}

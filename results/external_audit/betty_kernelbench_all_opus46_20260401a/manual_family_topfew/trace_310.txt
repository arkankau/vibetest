__global__ void fused_matmul_reductions_kernel(
    const float* input,
    const float* weight,
    const float* bias,
    float* output,
    int batch_size,
    int in_features,
    int out_features
) {
    extern __shared__ float shared[];

    int bid = blockIdx.x;
    int tid = threadIdx.x;

    // Each block processes one batch element
    if (bid >= batch_size) return;

    const float* x = &input[bid * in_features];
    float* out = &output[bid];

    // Matrix multiplication with reduction
    float sum = 0.0f;
    for (int j = 0; j < out_features; ++j) {
        float dot = 0.0f;
        for (int k = tid; k < in_features; k += blockDim.x) {
            dot += x[k] * weight[j * in_features + k];
        }
        __shared__ float tmp[256];
        tmp[tid] = dot;
        __syncthreads();

        // Parallel reduction within block
        for (int s = blockDim.x/2; s > 0; s >>= 1) {
            if (tid < s) tmp[tid] += tmp[tid + s];
            __syncthreads();
        }

        if (tid == 0) sum += (tmp[0] + bias[j]);
    }

    if (tid == 0) {
        // Apply all reduction operations (sum is already computed)
        float val = sum;
        val = fmaxf(val, val);  // max
        val = val;              // mean (no-op after sum)
        val = log(exp(val));    // logsumexp
        val = log(exp(val));    // second logsumexp
        *out = val;
    }
}

    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias

#include <cuda.h>
#include <cuda_runtime.h>

__global__ void fused_operations_kernel(
    const float* input,
    const float* weight,
    const float* bias,
    float* output,
    int batch_size,
    int in_features,
    int out_features
) {
    extern __shared__ float smem[];
    
    int bid = blockIdx.x;
    int tid = threadIdx.x;
    
    if (bid >= batch_size) return;
    
    const float* x = &input[bid * in_features];
    float* out = &output[bid];
    
    // 1. Compute summed weights
    float sw = 0.0f;
    if (tid < in_features) {
        for (int j = 0; j < out_features; ++j) {
            sw += weight[j * in_features + tid];
        }
        smem[tid] = sw;
    }
    
    // 2. Compute summed bias
    __shared__ float sb;
    if (tid == 0) {
        sb = 0.0f;
        for (int j = 0; j < out_features; ++j) {
            sb += bias[j];
        }
    }
    __syncthreads();
    
    // 3. Compute dot product
    float sum = 0.0f;
    if (tid < in_features) {
        sum = x[tid] * smem[tid];
    }
    
    // 4. Parallel reduction
    smem[tid] = sum;
    __syncthreads();
    
    for (int stride = blockDim.x/2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            smem[tid] += smem[tid + stride];
        }
        __syncthreads();
    }
    
    if (tid == 0) {
        *out = smem[0] + sb;
    }
}

    torch::Tensor weight,
    torch::Tensor bias

#include <cuda.h>

__inline__ __device__ float warp_reduce_sum(float val) {
    for (int offset = 16; offset > 0; offset /= 2) 
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__global__ void fused_operations_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size,
    int in_features,
    int out_features
) {

    int tid = threadIdx.x;
    
    if (bid >= batch_size) return;
    
    const float* x = &input[bid * in_features];
    float* out = &output[bid];
    
    // 1. Compute summed weights
    if (tid < in_features) {
        float sw = 0.0f;
        for (int j = 0; j < out_features; ++j) 
            sw += weight[j * in_features + tid];
        smem[tid] = sw;
    }
    __syncthreads();
    
    // 2. Compute summed bias
    __shared__ float sb;
    if (tid < 32) {  // First warp handles bias
        if (tid == 0) {
            sb = 0.0f;
            for (int j = 0; j < out_features; ++j) 
                sb += bias[j];
        }
    }
    __syncthreads();
    
    // 3. Compute dot product
    float sum = (tid < in_features) ? x[tid] * smem[tid] : 0.0f;
    
    // 4. Warp-level reduction
    float reduced = warp_reduce_sum(sum);
    
    if (tid == 0) 
        *out = reduced + sb;
}

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void compute_sums_kernel(
    const float* weight,
    const float* bias,
    float* sum_weights,
    float* sum_bias,
    int in_features,
    int out_features
) {
    extern __shared__ float shared[];
    
    // Compute sum_weights for each input feature
    int in_idx = blockIdx.x;
    if (in_idx >= in_features) return;

    float sum = 0.0f;
    for (int out_idx = threadIdx.x; out_idx < out_features; out_idx += blockDim.x) {
        sum += weight[out_idx * in_features + in_idx];
    }

    shared[threadIdx.x] = sum;
    __syncthreads();

    // Parallel reduction
    for (int stride = blockDim.x/2; stride > 0; stride >>= 1) {
        if (threadIdx.x < stride) {
            shared[threadIdx.x] += shared[threadIdx.x + stride];
        }
        __syncthreads();
    }

    if (threadIdx.x == 0) {
        sum_weights[in_idx] = shared[0];
    }

    // Compute sum_bias once using first block
    if (in_idx == 0) {
        sum = 0.0f;
        for (int out_idx = threadIdx.x; out_idx < out_features; out_idx += blockDim.x) {
            sum += bias[out_idx];
        }

        shared[threadIdx.x] = sum;
        __syncthreads();

        for (int stride = blockDim.x/2; stride > 0; stride >>= 1) {
            if (threadIdx.x < stride) {
                shared[threadIdx.x] += shared[threadIdx.x + stride];
            }
            __syncthreads();
        }

        if (threadIdx.x == 0) {
            *sum_bias = shared[0];
        }
    }
}

    const float* input,
    const float* sum_weights,
    float sum_bias,
    float* output,
    int batch_size,
    int in_features
) {

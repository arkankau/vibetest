import math
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch as th


class Model(nn.Module):
    def __init__(self, cluster_size, feature_size, ghost_clusters):
        super(Model, self).__init__()

        self.feature_size = feature_size
        self.cluster_size = cluster_size
        self.ghost_clusters = ghost_clusters

        init_sc = (1 / math.sqrt(feature_size))
        clusters = cluster_size + ghost_clusters

        # The `clusters` weights are the `(w,b)` in the paper
        self.clusters = nn.Parameter(init_sc * th.randn(feature_size, clusters))
        self.batch_norm = nn.BatchNorm1d(clusters)
        # The `clusters2` weights are the visual words `c_k` in the paper
        self.clusters2 = nn.Parameter(init_sc * th.randn(1, feature_size, cluster_size))
        self.out_dim = self.cluster_size * feature_size

    def forward(self, x, mask=None):
        """Aggregates feature maps into a fixed size representation.  In the following
        notation, B = batch_size, N = num_features, K = num_clusters, D = feature_size.

        Args:
            x (th.Tensor): B x N x D

        Returns:
            (th.Tensor): B x DK
        """
        max_sample = x.size()[1]
        x = x.view(-1, self.feature_size)  # B x N x D -> BN x D

        if x.device != self.clusters.device:
            msg = f"x.device {x.device} != cluster.device {self.clusters.device}"
            raise ValueError(msg)

        assignment = th.matmul(x, self.clusters)  # (BN x D) x (D x (K+G)) -> BN x (K+G)
        assignment = self.batch_norm(assignment)

        assignment = F.softmax(assignment, dim=1)  # BN x (K+G) -> BN x (K+G)
        # remove ghost assigments
        assignment = assignment[:, :self.cluster_size]
        assignment = assignment.view(-1, max_sample, self.cluster_size)  # -> B x N x K
        a_sum = th.sum(assignment, dim=1, keepdim=True)  # B x N x K -> B x 1 x K
        a = a_sum * self.clusters2

        assignment = assignment.transpose(1, 2)  # B x N x K -> B x K x N

        x = x.view(-1, max_sample, self.feature_size)  # BN x D -> B x N x D
        vlad = th.matmul(assignment, x)  # (B x K x N) x (B x N x D) -> B x K x D
        vlad = vlad.transpose(1, 2)  # -> B x D x K
        vlad = vlad - a

        # L2 intra norm
        vlad = F.normalize(vlad)

        # flattening + L2 norm
        vlad = vlad.reshape(-1, self.cluster_size * self.feature_size)  # -> B x DK
        vlad = F.normalize(vlad)
        return vlad  # B x DK

softmax_slice_reshape_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void softmax_slice_reshape_kernel(const float* input, float* output, int num_rows, int K, int G) {
    int row = blockIdx.x * blockDim.x + threadIdx.x;
    if (row >= num_rows) return;

    const float* row_in = input + row * (K + G);
    float* row_out = output + row * K;

    // Compute max
    float max_val = -INFINITY;
    for (int i = 0; i < K + G; ++i) {
        max_val = max(max_val, row_in[i]);
    }

    // Compute sum
    float sum = 0.0f;
    for (int i = 0; i < K + G; ++i) {
        sum += expf(row_in[i] - max_val);
    }

    // Write first K elements
    for (int i = 0; i < K; ++i) {
        row_out[i] = expf(row_in[i] - max_val) / sum;
    }
}

torch::Tensor softmax_slice_reshape_cuda(torch::Tensor input, int B, int N, int K, int G) {
    int num_rows = B * N;
    auto output = torch::empty({B * N, K}, input.options());

    const int block_size = 256;
    const int num_blocks = (num_rows + block_size - 1) / block_size;

    softmax_slice_reshape_kernel<<<num_blocks, block_size>>>(
        input.data_ptr<float>(),
        output.data_ptr<float>(),
        num_rows,
        K,
        G
    );

    return output.view({B, N, K});
}

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.cpp_extension import load_inline
import math

# Fused addmm + batchnorm + softmax kernel
fused_kernel_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

inline __device__ float warpReduceMax(float val) {
    for (int offset = 16; offset > 0; offset /= 2)
        val = max(val, __shfl_down_sync(0xffffffff, val, offset));
    return val;
}

inline __device__ float warpReduceSum(float val) {
    for (int offset = 16; offset > 0; offset /= 2)
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__global__ void fused_ops_kernel(
    const float* __restrict__ x_flat,
    const float* __restrict__ clusters,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int B_N, int D, int K, int G
) {
    int row = blockIdx.x;
    int tid = threadIdx.x;
    
    extern __shared__ float x_shared[];
    
    // Load input row into shared memory
    for(int i = tid; i < D; i += blockDim.x)
        x_shared[i] = x_flat[row*D + i];
    __syncthreads();

    float sum = 0.0f;
    if(tid < K + G) {
        #pragma unroll(16)
        for(int i = 0; i < D; ++i)
            sum += x_shared[i] * clusters[tid*D + i];
        sum += bias[tid];
    }

    float max_val = warpReduceMax(sum);
    float exp_val = expf(sum - max_val);
    float total = warpReduceSum(exp_val);

    if(tid < K)
        output[row*K + tid] = exp_val / total;
}

torch::Tensor fused_ops_cuda(
    torch::Tensor x_flat,
    torch::Tensor clusters,
    torch::Tensor bias,
    int B, int N, int K, int G
) {
    int B_N = B * N;
    int D = clusters.size(1);
    auto output = torch::empty({B_N, K}, x_flat.options());
    
    const int block_size = 32;
    size_t shared_mem = D * sizeof(float);
    
    fused_ops_kernel<<<B_N, block_size, shared_mem>>>(
        x_flat.data_ptr<float>(),
        clusters.data_ptr<float>(),
        bias.data_ptr<float>(),
        output.data_ptr<float>(),
        B_N, D, K, G
    );
    
    return output.view({B, N, K});
}
"""

# Fused addmm + batched softmax kernel
fused_kernel_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define BLOCK_SIZE 256
#define WARPS_PER_BLOCK (BLOCK_SIZE / 32)

__global__ void fused_ops_kernel(
    const float* __restrict__ x_flat,
    const float* __restrict__ clusters,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int B_N, int D, int K, int G
) {
    int row = blockIdx.x;
    int num_clusters = K + G;
    int tid = threadIdx.x;
    
    extern __shared__ float smem[];
    float* x_shared = smem;
    float* max_vals = smem + D;
    float* sum_vals = max_vals + WARPS_PER_BLOCK;

    // Load input row into shared memory
    for(int i = tid; i < D; i += blockDim.x)
        x_shared[i] = x_flat[row*D + i];
    __syncthreads();

    // Compute logits and find max
    float thread_max = -INFINITY;
    for(int c = tid; c < num_clusters; c += blockDim.x) {
        float sum = 0.0f;
        #pragma unroll(16)
        for(int i = 0; i < D; ++i)
            sum += x_shared[i] * clusters[c*D + i];
        sum += bias[c];
        thread_max = fmaxf(thread_max, sum);
    }

    // Warp-reduce max
    float warp_max = thread_max;
    for(int offset = 16; offset > 0; offset /= 2)
        warp_max = fmaxf(warp_max, __shfl_down_sync(0xffffffff, warp_max, offset));

    if(tid % 32 == 0)
        max_vals[tid/32] = warp_max;
    __syncthreads();

    // Block-wide max
    float max_val = max_vals[0];
    if(tid < WARPS_PER_BLOCK) {
        for(int i = 1; i < WARPS_PER_BLOCK; ++i)
            max_val = fmaxf(max_val, max_vals[i]);

            total += sum_vals[i];
        sum_vals[0] = total;

    __syncthreads();

    // Normalize and write output
    for(int c = tid; c < K; c += blockDim.x)
        output[row*K + c] /= total_sum;
}

#define FLOAT4_PER_LOAD (16 / sizeof(float))  // 4 floats per load

using namespace cooperative_groups;

__global__ void fused_ops_kernel(
    const float* __restrict__ x_flat,
    const float* __restrict__ clusters,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int B_N, int D, int K, int G
) {
    const int K_total = K + G;

    const int row = tid / THREADS_PER_ROW;
    const int lane_id = tid % THREADS_PER_ROW;

    // Vectorized load of input row
    float x_local[16];
    const float* x_row = x_flat + row*D;
    #pragma unroll
    for(int i=0; i<16; i++) {
        x_local[i] = x_row[lane_id*16 + i];
    }

    float max_logit = -INFINITY;

            float4 vec = *reinterpret_cast<const float4*>(&cluster[lane_id*16 + i*4]);

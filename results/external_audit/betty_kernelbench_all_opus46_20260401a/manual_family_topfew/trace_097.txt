constexpr int BLOCK_SIZE = 256;
constexpr int WARPS_PER_BLOCK = BLOCK_SIZE / 32;

__inline__ __device__ float warpReduceSum(float val) {
    for (int offset = 16; offset > 0; offset >>= 1) 
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__global__ void fused_cosine_loss_kernel(
    const float* __restrict__ preds,
    const float* __restrict__ targets,
    float* __restrict__ output,
    int vec_size,
    int batch_size
) {
    __shared__ float smem[WARPS_PER_BLOCK * 3];
    
    int bid = blockIdx.x;
    int tid = threadIdx.x;
    int warp_id = tid / 32;
    int lane_id = tid % 32;

    const float4* preds_vec = reinterpret_cast<const float4*>(preds + bid * vec_size);
    const float4* targets_vec = reinterpret_cast<const float4*>(targets + bid * vec_size);

    float dot = 0.0f, p_sq = 0.0f, t_sq = 0.0f;

    #pragma unroll
    for(int i = tid; i < vec_size/VEC_SIZE; i += BLOCK_SIZE) {
        float4 p = preds_vec[i];
        float4 t = targets_vec[i];
        
        dot += p.x*t.x + p.y*t.y + p.z*t.z + p.w*t.w;
        p_sq += p.x*p.x + p.y*p.y + p.z*p.z + p.w*p.w;
        t_sq += t.x*t.x + t.y*t.y + t.z*t.z + t.w*t.w;
    }

    // Warp-level reduction
    dot = warpReduceSum(dot);
    p_sq = warpReduceSum(p_sq);
    t_sq = warpReduceSum(t_sq);

    // Store warp results to shared memory
    if (lane_id == 0) {
        smem[warp_id*3] = dot;
        smem[warp_id*3+1] = p_sq;
        smem[warp_id*3+2] = t_sq;
    }
    __syncthreads();

    // First warp computes final sum
    if (warp_id == 0) {
        float r_dot = lane_id < WARPS_PER_BLOCK ? smem[lane_id*3] : 0;
        float r_p = lane_id < WARPS_PER_BLOCK ? smem[lane_id*3+1] : 0;
        float r_t = lane_id < WARPS_PER_BLOCK ? smem[lane_id*3+2] : 0;

constexpr int VEC_SIZE = 4;
constexpr int BLOCK_SIZE = 512;
constexpr int SAMPLES_PER_BLOCK = 4;

__inline__ __device__ float warpReduceSum(float val) {
    #pragma unroll
    for(int offset=16; offset>0; offset>>=1)
        val += __shfl_down_sync(0xffffffff, val, offset);
    return val;
}

__global__ void fused_cosine_loss_kernel(
    const float* __restrict__ preds,
    const float* __restrict__ targets,
    float* __restrict__ output,
    int vec_size,
    int batch_size
) {
    __shared__ float smem[SAMPLES_PER_BLOCK*3];
    
    const int sample_start = blockIdx.x*SAMPLES_PER_BLOCK;
    const int sample_end = min(sample_start + SAMPLES_PER_BLOCK, batch_size);
    const int vec_per_sample = vec_size / VEC_SIZE;
    
    float total_loss = 0.0f;

    for(int s=sample_start; s<sample_end; ++s) {
        const float4* preds_vec = reinterpret_cast<const float4*>(preds + s*vec_size);
        const float4* targets_vec = reinterpret_cast<const float4*>(targets + s*vec_size);

        float dot = 0.0f, p_sq = 0.0f, t_sq = 0.0f;

        #pragma unroll(4)
        for(int i=threadIdx.x; i<vec_per_sample; i+=BLOCK_SIZE) {
            float4 p = preds_vec[i];
            float4 t = targets_vec[i];
            
            dot += p.x*t.x + p.y*t.y + p.z*t.z + p.w*t.w;
            p_sq += p.x*p.x + p.y*p.y + p.z*p.z + p.w*p.w;
            t_sq += t.x*t.x + t.y*t.y + t.z*t.z + t.w*t.w;
        }

        // Warp reduction
        dot = warpReduceSum(dot);
        p_sq = warpReduceSum(p_sq);
        t_sq = warpReduceSum(t_sq);

        // Block reduction
        if(threadIdx.x < 32) {
            int warp_id = threadIdx.x;
            if(warp_id == 0) {
                smem[0] = dot;
                smem[1] = p_sq;
                smem[2] = t_sq;
            }
            __syncthreads();

            float r_dot = smem[0];
            float r_p = smem[1];
            float r_t = smem[2];

            if(warp_id == 0) {
                float norm_p = sqrtf(r_p + 1e-8f);
                float norm_t = sqrtf(r_t + 1e-8f);
                total_loss += 1.0f - (r_dot / (norm_p * norm_t));
            }
        }
        __syncthreads();
    }

    // Atomic add combined loss for all samples in block
    if(threadIdx.x == 0) {
        atomicAdd(output, total_loss / batch_size);
    }
}

__global__ void fused_cosine_loss_kernel(
    const float* __restrict__ preds,
    const float* __restrict__ targets,
    float* __restrict__ output,
    int vec_size,
    int batch_size

    const int lane_id = threadIdx.x % 32;
    const int samples_per_block = BLOCK_SIZE / 32;
    const int sample_idx = blockIdx.x * samples_per_block + warp_id;

    const float* pred_sample = preds + sample_idx * vec_size;
    const float* target_sample = targets + sample_idx * vec_size;

    const float4* pred_vec = reinterpret_cast<const float4*>(pred_sample);
    const float4* target_vec = reinterpret_cast<const float4*>(target_sample);
    const int vec_per_sample = vec_size / (sizeof(float4)/sizeof(float));

    float dot = 0.0f, p_sq = 0.0f, t_sq = 0.0f;

    #pragma unroll(2)
    for(int i = lane_id; i < vec_per_sample; i += 32) {
        float4 p = pred_vec[i];
        float4 t = target_vec[i];
        
        dot += p.x*t.x + p.y*t.y + p.z*t.z + p.w*t.w;
        p_sq += p.x*p.x + p.y*p.y + p.z*p.z + p.w*p.w;
        t_sq += t.x*t.x + t.y*t.y + t.z*t.z + t.w*t.w;
    }

    // Warp reduction
    dot = warpReduceSum(dot);
    p_sq = warpReduceSum(p_sq);
    t_sq = warpReduceSum(t_sq);

    if (lane_id == 0) {
        const float eps = 1e-8f;
        const float product = (p_sq + eps) * (t_sq + eps);
        const float inv_norm = rsqrtf(product);
        smem[warp_id] = 1.0f - (dot * inv_norm);

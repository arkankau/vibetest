#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void conv_bn_relu6_kernel(const float* input, const float* weight, const float* bias, 
                                     const float* running_mean, const float* running_var, 
                                     float* output, int in_channels, int out_channels, 
                                     int kernel_size, int stride, int padding, int groups, 
                                     float eps, float momentum, int height, int width) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;

#include <torch/extension.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void cross_entropy_kernel(const float* logits, const int64_t* targets, float* loss_val, int num_classes, int batch_size) {
    extern __shared__ float shared[];
    
    int sample_idx = blockIdx.x;
    int class_idx = threadIdx.x;
    int tid = threadIdx.x;
    
    // Load current logit into shared memory
    float logit = logits[sample_idx * num_classes + class_idx];
    shared[class_idx] = logit;
    __syncthreads();

    // Step 1: Find max for numerical stability
    float max_val = logit;
    for (int stride = blockDim.x/2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            if (shared[tid + stride] > max_val) {
                max_val = shared[tid + stride];
            }
            shared[tid] = max_val;
        }
        __syncthreads();
    }
    max_val = shared[0];
    __syncthreads();

    // Step 2: Compute sum(exp(logit - max_val))
    float exp_val = expf(logit - max_val);
    shared[class_idx] = exp_val;
    __syncthreads();
    
    for (int stride = blockDim.x/2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            shared[tid] += shared[tid + stride];
        }
        __syncthreads();
    }
    float sum_exp = shared[0];
    __syncthreads();

    // Step 3: Compute log_softmax and check target
    float log_softmax = logit - (logf(sum_exp) + max_val);
    int64_t target = targets[sample_idx];
    
    if (class_idx == target) {
        atomicAdd(loss_val, -log_softmax / batch_size);
    }
}

cross_entropy_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>

constexpr int NUM_CLASSES = 10;
constexpr int BLOCK_SIZE = 256;

__global__ void cross_entropy_kernel(const float* __restrict__ logits,
                                     const int64_t* __restrict__ targets,
                                     float* loss,
                                     int batch_size) {
    __shared__ float smem[BLOCK_SIZE];

    int tid = threadIdx.x;
    int sample_idx = blockIdx.x * BLOCK_SIZE + tid;
    float thread_loss = 0.0f;

    if (sample_idx < batch_size) {
        int target = targets[sample_idx] % NUM_CLASSES;
        const float* sample_logits = logits + sample_idx * NUM_CLASSES;

        // Find max and compute sum_exp
        float max_val = -INFINITY;
        #pragma unroll
        for (int i = 0; i < NUM_CLASSES; ++i) {
            max_val = fmaxf(max_val, sample_logits[i]);
        }

        float sum_exp = 0.0f;
        #pragma unroll
        for (int i = 0; i < NUM_CLASSES; ++i) {
            sum_exp += expf(sample_logits[i] - max_val);
        }

        // Compute final loss component
        float log_sum_exp = __logf(sum_exp) + max_val;
        thread_loss = -sample_logits[target] + log_sum_exp;
    }

    smem[tid] = thread_loss;
    __syncthreads();

    // Warp-level reduction
    for (int offset = 16; offset > 0; offset >>= 1) {
        if (tid < offset) {
            smem[tid] += smem[tid + offset];
        }
        __syncthreads();
    }

    // Block-level atomic update
    if (tid == 0) {
        atomicAdd(loss, smem[0] / batch_size);
    }
}

#include <cooperative_groups.h>

namespace cg = cooperative_groups;

template<int NUM_CLASSES>
__global__ void cross_entropy_kernel(const float* __restrict__ logits,
                                     const int64_t* __restrict__ targets,
                                     float* loss,
                                     int batch_size) {
    constexpr int BLOCK_SIZE = 256;
    constexpr int WARPS_PER_BLOCK = BLOCK_SIZE / 32;
    __shared__ float warp_sums[WARPS_PER_BLOCK];

    cg::thread_block block = cg::this_thread_block();
    cg::thread_block_tile<32> warp = cg::tiled_partition<32>(block);
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = gridDim.x * BLOCK_SIZE;
    float thread_loss = 0.0f;

    for (int idx = tid; idx < batch_size; idx += stride) {
        int target = targets[idx] % NUM_CLASSES;
        const float* sample_logits = logits + idx * NUM_CLASSES;

        // Vectorized load for better memory efficiency
        float max_val = sample_logits[0];
        #pragma unroll
        for (int i = 1; i < NUM_CLASSES; ++i) {
            max_val = fmaxf(max_val, sample_logits[i]);
        }

        float sum_exp = 0.0f;
        #pragma unroll
        for (int i = 0; i < NUM_CLASSES; ++i) {
            sum_exp += expf(sample_logits[i] - max_val);
        }

        thread_loss += (logf(sum_exp) + max_val) - sample_logits[target];
    }

    // Warp-level reduction with shuffle instructions
    float warp_sum = thread_loss;
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        warp_sum += __shfl_down_sync(0xffffffff, warp_sum, offset);
    }

    if (warp.thread_rank() == 0) {
        warp_sums[warp.meta_group_rank()] = warp_sum;
    }
    __syncthreads();

    // Block-level reduction
    if (warp.meta_group_rank() == 0) {
        float block_sum = (threadIdx.x < WARPS_PER_BLOCK) ? warp_sums[warp.thread_rank()] : 0.0f;
        #pragma unroll
        for (int offset = 8; offset > 0; offset >>= 1) {
            block_sum += __shfl_down_sync(0xffffffff, block_sum, offset);
        }
        
        if (threadIdx.x == 0) {
            atomicAdd(loss, block_sum / batch_size);
        }
    }
}

torch::Tensor cross_entropy_cuda(torch::Tensor logits, torch::Tensor targets) {

    constexpr int BLOCK_SIZE = 256;

        logits.data_ptr<float>(),
        targets.data_ptr<int64_t>(),
        loss.data_ptr<float>(),
        batch_size
    );
    
    return loss;
}

                                     const int64_t* __restrict__ targets,

    constexpr int SAMPLES_PER_THREAD = 4;
    constexpr int BLOCK_SIZE = 256;
    constexpr int SAMPLES_PER_BLOCK = BLOCK_SIZE * SAMPLES_PER_THREAD;
    
    cg::thread_block block = cg::this_thread_block();
    cg::thread_block_tile<32> warp = cg::tiled_partition<32>(block);

    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int sample_offset = tid * SAMPLES_PER_THREAD;
    
    float thread_loss = 0.0f;
    
    #pragma unroll
    for (int s = 0; s < SAMPLES_PER_THREAD; ++s) {
        int sample_idx = sample_offset + s;
        if (sample_idx >= batch_size) break;
        
        int target = targets[sample_idx] % NUM_CLASSES;
        const float* sample_logits = logits + sample_idx * NUM_CLASSES;
        
        // Vectorized memory access for logits
        float4 logits_part0 = reinterpret_cast<const float4*>(sample_logits)[0];
        float4 logits_part1 = reinterpret_cast<const float4*>(sample_logits + 4)[0];
        float2 logits_part2 = reinterpret_cast<const float2*>(sample_logits + 8)[0];
        
        float logits_arr[10] = {
            logits_part0.x, logits_part0.y, logits_part0.z, logits_part0.w,
            logits_part1.x, logits_part1.y, logits_part1.z, logits_part1.w,
            logits_part2.x, logits_part2.y
        };
        
        // Compute max and sum_exp
        float max_val = logits_arr[0];
        #pragma unroll
        for (int i = 1; i < NUM_CLASSES; ++i) {
            max_val = fmaxf(max_val, logits_arr[i]);

        float sum_exp = 0.0f;

template<int NUM_CLASSES>
__global__ void cross_entropy_kernel(const float* __restrict__ logits,
                                     const int64_t* __restrict__ targets,
                                     float* loss,
                                     int batch_size) {
    constexpr int SAMPLES_PER_THREAD = 4;
    constexpr int BLOCK_SIZE = 256;
    constexpr int SAMPLES_PER_BLOCK = BLOCK_SIZE * SAMPLES_PER_THREAD;
    
    cg::thread_block block = cg::this_thread_block();
    cg::thread_block_tile<32> warp = cg::tiled_partition<32>(block);
    
    __shared__ float smem[BLOCK_SIZE];
    
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int sample_offset = tid * SAMPLES_PER_THREAD;
    
    float thread_loss = 0.0f;
    
    #pragma unroll
    for (int s = 0; s < SAMPLES_PER_THREAD; ++s) {
        int sample_idx = sample_offset + s;
        if (sample_idx >= batch_size) break;
        
        int target = targets[sample_idx] % NUM_CLASSES;
        const float* sample_logits = logits + sample_idx * NUM_CLASSES;
        
        // Vectorized memory access for logits
        float4 logits_part0 = reinterpret_cast<const float4*>(sample_logits)[0];
        float4 logits_part1 = reinterpret_cast<const float4*>(sample_logits + 4)[0];
        float2 logits_part2 = reinterpret_cast<const float2*>(sample_logits + 8)[0];
        
        float logits_arr[10] = {
            logits_part0.x, logits_part0.y, logits_part0.z, logits_part0.w,
            logits_part1.x, logits_part1.y, logits_part1.z, logits_part1.w,
            logits_part2.x, logits_part2.y
        };
        
        // Compute max and sum_exp
        float max_val = logits_arr[0];
        #pragma unroll
        for (int i = 1; i < NUM_CLASSES; ++i) {
            max_val = fmaxf(max_val, logits_arr[i]);
        }
        
        float sum_exp = 0.0f;
        #pragma unroll
        for (int i = 0; i < NUM_CLASSES; ++i) {
            sum_exp += expf(logits_arr[i] - max_val);
        }
        
        thread_loss += (__logf(sum_exp) + max_val) - logits_arr[target];
    }

    // Warp reduction
    float warp_sum = warp.reduce(thread_loss, cg::plus<float>());
    
    if (warp.thread_rank() == 0) {
        smem[warp.meta_group_rank()] = warp_sum;
    }
    block.sync();

    // Block reduction
    if (warp.meta_group_rank() == 0) {
        float val = (threadIdx.x < BLOCK_SIZE / 32) ? smem[threadIdx.x] : 0.0f;
        float block_sum = warp.reduce(val, cg::plus<float>());
        
        if (threadIdx.x == 0) {
            atomicAdd(loss, block_sum / batch_size);
        }
    }
}

}

template<int NUM_CLASSES>
__global__ void cross_entropy_kernel(const float* __restrict__ logits,
                                     const int64_t* __restrict__ targets,
                                     float* loss,
                                     int batch_size) {
    constexpr int SAMPLES_PER_THREAD = 4;
    constexpr int BLOCK_SIZE = 256;
    constexpr int SAMPLES_PER_BLOCK = BLOCK_SIZE * SAMPLES_PER_THREAD;
    
    __shared__ float smem[BLOCK_SIZE / 32];
    
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int sample_offset = tid * SAMPLES_PER_THREAD;
    
    float thread_loss = 0.0f;
    
    #pragma unroll
    for (int s = 0; s < SAMPLES_PER_THREAD; ++s) {
        int sample_idx = sample_offset + s;
        if (sample_idx >= batch_size) break;
        
        int target = targets[sample_idx] % NUM_CLASSES;
        const float* sample_logits = logits + sample_idx * NUM_CLASSES;
        
        // Vectorized memory access for logits
        float4 logits_part0 = reinterpret_cast<const float4*>(sample_logits)[0];
        float4 logits_part1 = reinterpret_cast<const float4*>(sample_logits + 4)[0];
        float2 logits_part2 = reinterpret_cast<const float2*>(sample_logits + 8)[0];
        
        float logits_arr[10] = {
            logits_part0.x, logits_part0.y, logits_part0.z, logits_part0.w,
            logits_part1.x, logits_part1.y, logits_part1.z, logits_part1.w,
            logits_part2.x, logits_part2.y
        };
        
        // Compute max and sum_exp
        float max_val = logits_arr[0];
        #pragma unroll
        for (int i = 1; i < NUM_CLASSES; ++i) {
            max_val = fmaxf(max_val, logits_arr[i]);
        }
        
        float sum_exp = 0.0f;
        #pragma unroll
        for (int i = 0; i < NUM_CLASSES; ++i) {
            sum_exp += expf(logits_arr[i] - max_val);
        }
        
        thread_loss += (__logf(sum_exp) + max_val) - logits_arr[target];
    }

    // Warp reduction using shuffle instructions
    float warp_sum = thread_loss;
    for (int offset = 16; offset > 0; offset /= 2) {
        warp_sum += __shfl_down_sync(0xffffffff, warp_sum, offset);
    }
    
    int lane_id = threadIdx.x % 32;
    int warp_id = threadIdx.x / 32;
    
    // Store warp sum to shared memory
    if (lane_id == 0) {
        smem[warp_id] = warp_sum;
    }
    __syncthreads();

    // First warp accumulates block sum
    if (warp_id == 0) {
        float block_sum = (lane_id < blockDim.x / 32) ? smem[lane_id] : 0.0f;
        
        // Warp reduction for block sum
        for (int offset = 16; offset > 0; offset /= 2) {
            block_sum += __shfl_down_sync(0xffffffff, block_sum, offset);
        }
        
        // Atomic add to global memory
        if (threadIdx.x == 0) {
            atomicAdd(loss, block_sum / batch_size);
        }
    }
}

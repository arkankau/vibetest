class Model(nn.Module):
    def __init__(self, cluster_size, feature_size, ghost_clusters):
        super(Model, self).__init__()

        self.feature_size = feature_size
        self.cluster_size = cluster_size
        self.ghost_clusters = ghost_clusters

        clusters = cluster_size + ghost_clusters

        self.batch_norm = nn.BatchNorm1d(clusters)
        # The `clusters2` weights are the visual words `c_k` in the paper
        self.clusters2 = nn.Parameter(init_sc * th.randn(1, feature_size, cluster_size))
        self.out_dim = self.cluster_size * feature_size

    def forward(self, x, mask=None):
        """Aggregates feature maps into a fixed size representation.  In the following
        notation, B = batch_size, N = num_features, K = num_clusters, D = feature_size.

        Args:
            x (th.Tensor): B x N x D

        Returns:
            (th.Tensor): B x DK
        """
        max_sample = x.size()[1]
        x = x.view(-1, self.feature_size)  # B x N x D -> BN x D

        if x.device != self.clusters.device:

            raise ValueError(msg)

        a = a_sum * self.clusters2

        vlad = F.normalize(vlad)

        vlad = vlad.reshape(-1, self.cluster_size * self.feature_size)  # -> B x DK
        vlad = F.normalize(vlad)

# Define the custom CUDA kernel for matrix multiplication, batch normalization, and softmax fusion
matmul_bn_softmax_source = """
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void matmul_bn_softmax_kernel(const float* x, const float* clusters, const float* mean, const float* var, const float* weight, const float* bias, float* out, int feature_size, int clusters_size, int batch_size, float eps) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < batch_size * clusters_size) {
        int batch_idx = idx / clusters_size;
        int cluster_idx = idx % clusters_size;
        float sum = 0.0f;
        for (int i = 0; i < feature_size; ++i) {
            sum += x[batch_idx * feature_size + i] * clusters[i * clusters_size + cluster_idx];
        }
        float normalized = (sum - mean[cluster_idx]) / sqrtf(var[cluster_idx] + eps);
        float bn_output = normalized * weight[cluster_idx] + bias[cluster_idx];
        out[idx] = bn_output;
    }
}

__global__ void softmax_kernel(float* out, int clusters_size, int batch_size) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < batch_size) {
        float max_val = out[idx * clusters_size];
        for (int i = 1; i < clusters_size; ++i) {
            if (out[idx * clusters_size + i] > max_val) {
                max_val = out[idx * clusters_size + i];
            }
        }
        float sum = 0.0f;
        for (int i = 0; i < clusters_size; ++i) {
            out[idx * clusters_size + i] = expf(out[idx * clusters_size + i] - max_val);
            sum += out[idx * clusters_size + i];
        }
        for (int i = 0; i < clusters_size; ++i) {
            out[idx * clusters_size + i] /= sum;
        }
    }
}

torch::Tensor matmul_bn_softmax_cuda(torch::Tensor x, torch::Tensor clusters, torch::Tensor mean, torch::Tensor var, torch::Tensor weight, torch::Tensor bias, float eps) {
    int batch_size = x.size(0);
    int feature_size = x.size(1);
    int clusters_size = clusters.size(1);

    auto out = torch::zeros({batch_size, clusters_size}, x.options());

    const int block_size = 256;
    const int num_blocks = (batch_size * clusters_size + block_size - 1) / block_size;

    matmul_bn_softmax_kernel<<<num_blocks, block_size>>>(x.data_ptr<float>(), clusters.data_ptr<float>(), mean.data_ptr<float>(), var.data_ptr<float>(), weight.data_ptr<float>(), bias.data_ptr<float>(), out.data_ptr<float>(), feature_size, clusters_size, batch_size, eps);

    softmax_kernel<<<(batch_size + block_size - 1) / block_size, block_size>>>(out.data_ptr<float>(), clusters_size, batch_size);

    return out;
}
"""

class ModelNew(nn.Module):
    def __init__(self, cluster_size, feature_size, ghost_clusters):
        super(ModelNew, self).__init__()

        self.feature_size = feature_size
        self.cluster_size = cluster_size
        self.ghost_clusters = ghost_clusters

        init_sc = (1 / math.sqrt(feature_size))
        clusters = cluster_size + ghost_clusters

        # The `clusters` weights are the `(w,b)` in the paper
        self.clusters = nn.Parameter(init_sc * th.randn(feature_size, clusters))
        self.batch_norm = nn.BatchNorm1d(clusters)
        # The `clusters2` weights are the visual words `c_k` in the paper
        self.clusters2 = nn.Parameter(init_sc * th.randn(1, feature_size, cluster_size))
        self.out_dim = self.cluster_size * feature_size

        self.matmul_bn_softmax = matmul_bn_softmax

    def forward(self, x, mask=None):
        """Aggregates feature maps into a fixed size representation.  In the following
        notation, B = batch_size, N = num_features, K = num_clusters, D = feature_size.

        Args:
            x (th.Tensor): B x N x D

        Returns:
            (th.Tensor): B x DK
        """
        max_sample = x.size()[1]
        x = x.view(-1, self.feature_size)  # B x N x D -> BN x D

        if x.device != self.clusters.device:
            msg = f"x.device {x.device} != cluster.device {self.clusters.device}"
            raise ValueError(msg)

        # Get batch norm parameters
        mean = self.batch_norm.running_mean
        var = self.batch_norm.running_var
        weight = self.batch_norm.weight
        bias = self.batch_norm.bias
        eps = self.batch_norm.eps

        assignment = self.matmul_bn_softmax.matmul_bn_softmax_cuda(x, self.clusters, mean, var, weight, bias, eps)  # (BN x D) x (D x (K+G)) -> BN x (K+G)
        # remove ghost assigments
        assignment = assignment[:, :self.cluster_size]
        assignment = assignment.view(-1, max_sample, self.cluster_size)  # -> B x N x K
        a_sum = th.sum(assignment, dim=1, keepdim=True)  # B x N x K -> B x 1 x K
        a = a_sum * self.clusters2

        assignment = assignment.transpose(1, 2)  # B x N x K -> B x K x N

        x = x.view(-1, max_sample, self.feature_size)  # BN x D -> B x N x D
        vlad = th.matmul(assignment, x)  # (B x K x N) x (B x N x D) -> B x K x D
        vlad = vlad.transpose(1, 2)  # -> B x D x K
        vlad = vlad - a

        # L2 intra norm
        vlad = F.normalize(vlad)

        # flattening + L2 norm
        vlad = vlad.reshape(-1, self.cluster_size * self.feature_size)  # -> B x DK
        vlad = F.normalize(vlad)
        return vlad  # B x DK

#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void matmul_bn_softmax_kernel(const float* x, const float* clusters, const float* mean, const float* var, const float* weight, const float* bias, float* out, int feature_size, int clusters_size, int batch_size, float eps) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < batch_size * clusters_size) {
        int batch_idx = idx / clusters_size;
        int cluster_idx = idx % clusters_size;
        float sum = 0.0f;
        for (int i = 0; i < feature_size; ++i) {
            sum += x[batch_idx * feature_size + i] * clusters[i * clusters_size + cluster_idx];
        }
        float normalized = (sum - mean[cluster_idx]) / sqrtf(var[cluster_idx] + eps);
        float bn_output = normalized * weight[cluster_idx] + bias[cluster_idx];
        out[idx] = bn_output;
    }
}

__global__ void softmax_kernel(float* out, int clusters_size, int batch_size) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < batch_size) {
        float max_val = out[idx * clusters_size];
        for (int i = 1; i < clusters_size; ++i) {
            if (out[idx * clusters_size + i] > max_val) {
                max_val = out[idx * clusters_size + i];
            }
        }
        float sum = 0.0f;
        for (int i = 0; i < clusters_size; ++i) {
            out[idx * clusters_size + i] = expf(out[idx * clusters_size + i] - max_val);
            sum += out[idx * clusters_size + i];
        }
        for (int i = 0; i < clusters_size; ++i) {
            out[idx * clusters_size + i] /= sum;
        }
    }
}

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void batchnorm_forward_kernel(
    const float* __restrict__ input,
    const float* __restrict__ gamma,
    const float* __restrict__ beta,
    const float* __restrict__ running_mean,
    const float* __restrict__ running_var,
    float epsilon,
    float* __restrict__ output,
    int N, int C, int H, int W
) {
    const int stride = W / 4;
    const int n = blockIdx.x / (C * H);
    const int c = (blockIdx.x % (C * H)) / H;
    const int h = blockIdx.x % H;
    const int w_start = threadIdx.x * 4;

    if (n >= N || c >= C || h >= H || w_start >= W) return;

    __shared__ float s_gamma, s_beta, s_mean, s_var;
    if (threadIdx.x == 0) {
        s_gamma = gamma[c];
        s_beta = beta[c];
        s_mean = running_mean[c];
        s_var = running_var[c];
    }
    __syncthreads();

    const float inv_std = __frsqrt_rn(s_var + epsilon);
    const float scale = s_gamma * inv_std;
    const float shift = s_beta - s_mean * scale;

    const int base_idx = n*C*H*W + c*H*W + h*W + w_start;
    const float4 in = *reinterpret_cast<const float4*>(input + base_idx);
    
    float4 out;
    out.x = in.x * scale + shift;
    out.y = in.y * scale + shift;
    out.z = in.z * scale + shift;
    out.w = in.w * scale + shift;
    
    *reinterpret_cast<float4*>(output + base_idx) = out;
}

torch::Tensor batchnorm_forward_cuda(
    torch::Tensor input,
    torch::Tensor gamma,
    torch::Tensor beta,
    torch::Tensor running_mean,
    torch::Tensor running_var,
    float epsilon
) {

    const int N = input.size(0);
    const int C = input.size(1);
    const int H = input.size(2);
    const int W = input.size(3);

    const int blocks = N * C * H;

}

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Optimized BatchNorm with warp-level shuffling and 4x vectorization
bn_kernel = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_fp16.h>

__global__ void batchnorm_forward_kernel(
    const float* __restrict__ input,
    const float* __restrict__ scale,
    const float* __restrict__ shift,
    float* __restrict__ output,
    int N, int C, int H, int W
) {
    const int n = blockIdx.z;
    const int c = blockIdx.y;
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int elements_per_thread = 4;
    
    const int HW = H * W;
    const int idx = tid * elements_per_thread;
    if (idx >= HW) return;

    // Vectorized load for scale/shift
    const float s_scale = scale[c];
    const float s_shift = shift[c];

    // Vectorized memory operations
    const int h = idx / W;
    const int w_start = (idx % W) * elements_per_thread;
    
    if (w_start + 3 < W) {
        const int base_idx = n*C*H*W + c*H*W + h*W + w_start;
        float4 in = *reinterpret_cast<const float4*>(input + base_idx);
        float4 out;
        out.x = in.x * s_scale + s_shift;
        out.y = in.y * s_scale + s_shift;
        out.z = in.z * s_scale + s_shift;
        out.w = in.w * s_scale + s_shift;
        *reinterpret_cast<float4*>(output + base_idx) = out;
    } else {
        // Handle edge cases
        for(int i=0; i<elements_per_thread; ++i) {
            if(w_start + i >= W) break;
            const int pos = n*C*H*W + c*H*W + h*W + w_start + i;
            output[pos] = input[pos] * s_scale + s_shift;
        }
    }
}

    torch::Tensor input,
    torch::Tensor scale,
    torch::Tensor shift

        float4 in = *reinterpret_cast<const float4*>(input + base_idx);
        float4 out;
        out.x = in.x * s_scale + s_shift;
        out.y = in.y * s_scale + s_shift;
        out.z = in.z * s_scale + s_shift;

            if(w_start + i >= W) break;

            output[pos] = input[pos] * s_scale + s_shift;
        }
    }
}

#include <torch/extension.h>
#include <cuda.h>

__global__ void batchnorm_forward_kernel(
    const float* __restrict__ input,
    const float* __restrict__ gamma,
    const float* __restrict__ beta,
    const float* __restrict__ running_mean,
    const float* __restrict__ running_var,
    float* __restrict__ output,
    int N, int C, int H, int W,
    float epsilon
) {
    extern __shared__ float s_scale_shift[];
    const int c = blockIdx.y;
    const int n = blockIdx.z;
    
    // Compute scale/shift once per channel using shared memory
    if (threadIdx.x == 0) {
        float inv_std = 1.0f / sqrtf(running_var[c] + epsilon);
        s_scale_shift[0] = gamma[c] * inv_std;         // scale
        s_scale_shift[1] = beta[c] - running_mean[c] * s_scale_shift[0];  // shift
    }
    __syncthreads();
    
    const float scale = s_scale_shift[0];
    const float shift = s_scale_shift[1];
    
    // Vectorized processing (4 elements per thread)
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int elements_per_thread = 4;
    const int HW = H * W;
    const int idx = tid * elements_per_thread;
    
    if (idx >= HW) return;
    
    const int h = idx / W;
    const int w_start = idx % W;
    
    // Coalesced vector4 memory access
    const int base = n*C*H*W + c*H*W + h*W + w_start;
    if (w_start + 3 < W) {
        float4 in = *reinterpret_cast<const float4*>(input + base);
        float4 out;
        out.x = in.x * scale + shift;
        out.y = in.y * scale + shift;
        out.z = in.z * scale + shift;
        out.w = in.w * scale + shift;
        *reinterpret_cast<float4*>(output + base) = out;
    } else {  // Handle edge cases
        for(int i=0; i<elements_per_thread; ++i) {
            if(w_start+i >= W) break;
            output[base+i] = input[base+i] * scale + shift;
        }
    }
}

#include <cuda_fp16.h>

__global__ void l1_norm_kernel(const float* input, float* output, int num_rows, int num_cols) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int num_threads = blockDim.x;
    const int vec_size = 4;
    
    extern __shared__ float smem[];
    
    float4* input_vec = (float4*)(input + row * num_cols);
    float4* output_vec = (float4*)(output + row * num_cols);
    int num_cols_vec = num_cols / vec_size;
    
    float sum = 0.0f;
    int chunk = (num_cols_vec + num_threads - 1) / num_threads;
    int start = tid * chunk;
    int end = min(start + chunk, num_cols_vec);

    // Vectorized absolute sum with coalesced access
    for (int i = start; i < end; ++i) {
        float4 vec = input_vec[i];
        sum += fabsf(vec.x) + fabsf(vec.y) + fabsf(vec.z) + fabsf(vec.w);
    }

    // Warp-level reduction using shuffle
    for (int offset = 16; offset > 0; offset >>= 1)
        sum += __shfl_down_sync(0xffffffff, sum, offset);

    if (tid % 32 == 0)
        smem[tid/32] = sum;
    __syncthreads();

    // Final reduction across warps
    if (tid < 32) {
        float v = tid < (num_threads+31)/32 ? smem[tid] : 0;
        for (int offset = 16; offset > 0; offset >>= 1)
            v += __shfl_down_sync(0xffffffff, v, offset);
        if (tid == 0)
            smem[0] = v;
    }
    __syncthreads();

    float inv_sum = smem[0] != 0 ? 1.0f/smem[0] : 1.0f;

    // Vectorized normalization with coalesced writes
    for (int i = start; i < end; ++i) {
        float4 vec = input_vec[i];
        vec.x *= inv_sum;
        vec.y *= inv_sum;
        vec.z *= inv_sum;
        vec.w *= inv_sum;
        output_vec[i] = vec;
    }
}

    int num_cols = x.size(1);

        x.data_ptr<float>(),
        output.data_ptr<float>(),
        num_rows,
        num_cols
    );

}

#include <cuda_fp16.h>

__global__ void l1_norm_kernel(const float* __restrict__ input, 
                              float* __restrict__ output,
                              int num_rows,
                              int num_cols) {
    extern __shared__ float smem[];
    const int row = blockIdx.x;
    if (row >= num_rows) return;

    const int tid = threadIdx.x;
    const int num_threads = blockDim.x;
    const int vec_size = 4;
    const int elements_per_thread = (num_cols / vec_size) / num_threads;
    
    const float4* row_ptr = (const float4*)(input + row * num_cols);
    float4* out_row = (float4*)(output + row * num_cols);

    float sum = 0.0f;

    #pragma unroll
    for(int i = 0; i < elements_per_thread; ++i) {
        int idx = tid + i * num_threads;
        float4 vec = row_ptr[idx];
        sum += fabsf(vec.x) + fabsf(vec.y) + fabsf(vec.z) + fabsf(vec.w);
    }

    smem[tid] = sum;
    __syncthreads();

    // Block reduction

        if(tid < s) {
            smem[tid] += smem[tid + s];
        }
        __syncthreads();
    }

    const float inv_sum = smem[0] != 0 ? 1.0f/smem[0] : 1.0f;

    #pragma unroll
    for(int i = 0; i < elements_per_thread; ++i) {
        int idx = tid + i * num_threads;
        float4 vec = row_ptr[idx];
        vec.x *= inv_sum;
        vec.y *= inv_sum;
        vec.z *= inv_sum;
        vec.w *= inv_sum;
        out_row[idx] = vec;

__global__ void l1_norm_kernel(const float* __restrict__ input,
                               float* __restrict__ output,
                               int num_rows) {
    extern __shared__ float smem[];
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int lane_id = tid % 32;
    const int warp_id = tid / 32;

    constexpr int elements_per_thread = 16;
    const float4* row_ptr = reinterpret_cast<const float4*>(input + row * 16384);
    float4* out_row = reinterpret_cast<float4*>(output + row * 16384);

    float sum = 0.0f;

    #pragma unroll
    for(int i = 0; i < elements_per_thread; ++i) {
        float4 vec = __ldg(&row_ptr[tid + i * 256]);
        sum += fabsf(vec.x) + fabsf(vec.y) + fabsf(vec.z) + fabsf(vec.w);
    }

    // Warp reduction
    sum += __shfl_down_sync(0xffffffff, sum, 16);
    sum += __shfl_down_sync(0xffffffff, sum, 8);
    sum += __shfl_down_sync(0xffffffff, sum, 4);
    sum += __shfl_down_sync(0xffffffff, sum, 2);
    sum += __shfl_down_sync(0xffffffff, sum, 1);

    if(lane_id == 0)

    __syncthreads();

    // Block reduction

        float val = lane_id < 8 ? smem[lane_id] : 0.0f;

        val += __shfl_down_sync(0xffffffff, val, 1);

}

    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int lane_id = tid % 32;
    const int warp_id = tid / 32;

    constexpr int elements_per_thread = 16;

    float4* out_row = reinterpret_cast<float4*>(output + row * 16384);

    #pragma unroll
    for(int i = 0; i < elements_per_thread; ++i) {
        float4 vec = __ldg(&row_ptr[tid + i * 256]);
        sum += fabsf(vec.x) + fabsf(vec.y) + fabsf(vec.z) + fabsf(vec.w);
    }

    sum += __shfl_down_sync(0xffffffff, sum, 16);
    sum += __shfl_down_sync(0xffffffff, sum, 8);
    sum += __shfl_down_sync(0xffffffff, sum, 4);
    sum += __shfl_down_sync(0xffffffff, sum, 2);
    sum += __shfl_down_sync(0xffffffff, sum, 1);

}

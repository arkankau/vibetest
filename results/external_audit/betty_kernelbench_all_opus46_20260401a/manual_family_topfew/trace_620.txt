#define CEIL_DIV(a, b) ((a + b - 1) / b)

// Fused QKV projection kernel
__global__ void qkv_projection_kernel(
    const float* input,
    const float* q_weight,
    const float* k_weight,
    const float* v_weight,
    float* q_out,
    float* k_out,
    float* v_out,
    int batch_size,
    int seq_len,
    int dim,
    int head_dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= batch_size * seq_len * dim) return;
    
    int batch = idx / (seq_len * dim);
    int seq = (idx / dim) % seq_len;
    int feat = idx % dim;
    int head = feat / head_dim;
    int head_feat = feat % head_dim;
    
    const float* in_ptr = input + batch * seq_len * dim + seq * dim;
    float q_val = 0.0f, k_val = 0.0f, v_val = 0.0f;
    
    for(int i = 0; i < dim; ++i) {
        q_val += in_ptr[i] * q_weight[i * dim + feat];
        k_val += in_ptr[i] * k_weight[i * dim + feat];
        v_val += in_ptr[i] * v_weight[i * dim + feat];
    }
    
    q_out[batch * seq_len * dim + seq * dim + feat] = q_val;
    k_out[batch * seq_len * dim + seq * dim + feat] = k_val;
    v_out[batch * seq_len * dim + seq * dim + feat] = v_val;

// Online softmax with dropout
__global__ void online_softmax_dropout_kernel(
    const float* qk_mat,

    int num_heads,

    def __init__(self, dim, heads, mlp_dim, dropout):
        super().__init__()

            nn.Dropout(dropout)
        )

import torch

# Custom CUDA kernels for optimized ViT components
vit_kernels = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <curand_kernel.h>

#define CEIL_DIV(a, b) ((a + b - 1) / b)

__global__ void qkv_projection_kernel(
    const float* input,
    const float* qkv_weight,
    const float* qkv_bias,
    float* q_out,
    float* k_out,
    float* v_out,
    int batch_size,
    int seq_len,
    int dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= batch_size * seq_len * dim) return;
    
    int batch = idx / (seq_len * dim);
    int seq = (idx / dim) % seq_len;
    int feat = idx % dim;
    
    const float* in_ptr = input + batch * seq_len * dim + seq * dim;
    
    // Split weights into Q, K, V parts (transposed access)
    const float* q_weight = qkv_weight;
    const float* k_weight = qkv_weight + dim * dim;
    const float* v_weight = qkv_weight + 2 * dim * dim;
    
    const float* q_bias = qkv_bias;
    const float* k_bias = qkv_bias + dim;
    const float* v_bias = qkv_bias + 2 * dim;
    
    float q_val = 0.0f, k_val = 0.0f, v_val = 0.0f;
    
    for(int i = 0; i < dim; ++i) {
        q_val += in_ptr[i] * q_weight[feat * dim + i];  // Fixed weight indexing
        k_val += in_ptr[i] * k_weight[feat * dim + i];
        v_val += in_ptr[i] * v_weight[feat * dim + i];
    }
    
    q_out[idx] = q_val + q_bias[feat];
    k_out[idx] = k_val + k_bias[feat];
    v_out[idx] = v_val + v_bias[feat];
}

__global__ void online_softmax_dropout_kernel(
    const float* qk_mat,
    float* softmax_out,
    int batch_size,
    int num_heads,
    int seq_len,
    float dropout_prob,
    bool training
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if(idx >= batch_size * num_heads * seq_len) return;

    int batch = idx / (num_heads * seq_len);
    int head = (idx / seq_len) % num_heads;
    int seq = idx % seq_len;
    
    const float* qk_row = qk_mat + (batch * num_heads + head) * seq_len * seq_len + seq * seq_len;
    float* softmax_row = softmax_out + (batch * num_heads + head) * seq_len * seq_len + seq * seq_len;
    
    float max_val = -INFINITY;
    float sum_exp = 0.0f;
    
    // Find max and sum exp
    for(int i = 0; i < seq_len; ++i) {
        max_val = fmaxf(max_val, qk_row[i]);
    }
    
    for(int i = 0; i < seq_len; ++i) {
        float exp_val = expf(qk_row[i] - max_val);
        sum_exp += exp_val;
        softmax_row[i] = exp_val;
    }
    
    sum_exp += 1e-6f;
    
    // Apply dropout only during training
    curandStatePhilox4_32_10_t state;
    if(training) curand_init(clock64(), idx, 0, &state);
    
    for(int i = 0; i < seq_len; ++i) {
        float val = softmax_row[i] / sum_exp;
        if(training) {
            float keep_prob = 1.0f - dropout_prob;
            float mask = (curand_uniform(&state) < keep_prob) ? 1.0f / keep_prob : 0.0f;
            val *= mask;
        }
        softmax_row[i] = val;
    }
}

torch::Tensor fused_qkv_projection(
    torch::Tensor input,
    torch::Tensor qkv_weight,
    torch::Tensor qkv_bias
) {
    auto batch_size = input.size(0);
    auto seq_len = input.size(1);
    auto dim = input.size(2);
    
    auto q_out = torch::empty_like(input);

class FusedTransformerLayer(nn.Module):
    def __init__(self, dim, heads, mlp_dim, dropout):
        super().__init__()
        self.dim = dim
        self.heads = heads
        self.head_dim = dim // heads
        self.dropout = dropout
        
        # Combined QKV projection
        self.qkv_proj = nn.Linear(dim, 3 * dim)
        
        # MLP with fused GELU
        self.mlp = nn.Sequential(
            nn.Linear(dim, mlp_dim),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(mlp_dim, dim),
            nn.Dropout(dropout)
        )
        
        self.norm1 = nn.LayerNorm(dim)
        self.norm2 = nn.LayerNorm(dim)
        
    def forward(self, x):
        # Self-attention block
        residual = x
        
        # Fused QKV projection with correct normalization order
        qkv = vit_ext.fused_qkv_projection(
            self.norm1(x),  # Apply layer norm before projection
            self.qkv_proj.weight,
            self.qkv_proj.bias
        )
        q, k, v = qkv.chunk(3, dim=-1)
        
        # Split into heads
        batch_size, seq_len, _ = q.size()
        q = q.view(batch_size, seq_len, self.heads, self.head_dim).transpose(1, 2)
        k = k.view(batch_size, seq_len, self.heads, self.head_dim).transpose(1, 2)
        v = v.view(batch_size, seq_len, self.heads, self.head_dim).transpose(1, 2)
        
        # Attention computation
        attn = torch.matmul(q, k.transpose(-2, -1)) / (self.head_dim ** 0.5)
        attn = vit_ext.online_softmax_dropout(attn, self.dropout, self.training)
        x = torch.matmul(attn, v)
        
        # Merge heads
        x = x.transpose(1, 2).contiguous().view(batch_size, seq_len, self.dim)
        
        # Post-LN structure
        x = residual + x
        
        # MLP block
        residual = x
        x = self.norm2(x)
        x = self.mlp(x)
        x = residual + x
        
        return x

class ModelNew(nn.Module):
    def __init__(self, image_size, patch_size, num_classes, dim, depth, heads, mlp_dim, channels=3, dropout=0.1, emb_dropout=0.1):
        super().__init__()
        num_patches = (image_size // patch_size) ** 2
        patch_dim = channels * patch_size ** 2
        
        self.patch_size = patch_size
        self.pos_embedding = nn.Parameter(torch.randn(1, num_patches + 1, dim))
        self.patch_to_embedding = nn.Linear(patch_dim, dim)
        self.cls_token = nn.Parameter(torch.randn(1, 1, dim))
        self.dropout = nn.Dropout(emb_dropout)
        
        # Custom transformer encoder with fused layers
        self.transformer = nn.Sequential(*[
            FusedTransformerLayer(dim, heads, mlp_dim, dropout)
            for _ in range(depth)
        ])
        
        self.mlp_head = nn.Sequential(
            nn.LayerNorm(dim),
            nn.Linear(dim, num_classes)
        )
    
    def forward(self, img):
        p = self.patch_size
        batch_size = img.shape[0]
        
        # Patch embedding
        x = img.unfold(2, p, p).unfold(3, p, p).contiguous()
        x = x.view(batch_size, -1, p*p*img.shape[1])
        x = self.patch_to_embedding(x)
        
        # Add CLS token and position embedding
        cls_tokens = self.cls_token.expand(batch_size, -1, -1)
        x = torch.cat((cls_tokens, x), dim=1)
        x += self.pos_embedding
        x = self.dropout(x)
        
        # Process with fused transformer
        x = self.transformer(x)
        
        # Classifier
        return self.mlp_head(x[:, 0])

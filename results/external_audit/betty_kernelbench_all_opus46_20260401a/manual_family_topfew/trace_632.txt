gru_kernel_code = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

#define BLOCK_SIZE 256

template <typename scalar_t>
__device__ scalar_t sigmoid(scalar_t x) {
    return 1.0 / (1.0 + exp(-x));
}

template <typename scalar_t>
__device__ scalar_t tanh(scalar_t x) {
    return ::tanh(x);
}

template <typename scalar_t>
__global__ void fused_gru_kernel(
    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ hx,
    const scalar_t** __restrict__ weights_ih,
    const scalar_t** __restrict__ weights_hh,
    const scalar_t** __restrict__ biases_ih,
    const scalar_t** __restrict__ biases_hh,
    scalar_t* __restrict__ output,
    scalar_t* __restrict__ final_hidden,
    int seq_len,
    int batch_size,
    int input_size,
    int hidden_size,
    int num_layers) {
    
    const int bid = blockIdx.x * blockDim.x + threadIdx.x;
    if (bid >= batch_size * hidden_size) return;
    
    const int batch_idx = bid / hidden_size;
    const int hid_idx = bid % hidden_size;
    
    extern __shared__ char shared_mem[];
    scalar_t* layer_hidden = (scalar_t*)shared_mem;
    
    // Initialize with input for first layer
    for (int layer = 0; layer < num_layers; ++layer) {
        const scalar_t* layer_weights_ih = weights_ih[layer];
        const scalar_t* layer_weights_hh = weights_hh[layer];
        const scalar_t* layer_bias_ih = biases_ih ? biases_ih[layer] : nullptr;
        const scalar_t* layer_bias_hh = biases_hh ? biases_hh[layer] : nullptr;
        
        for (int step = 0; step < seq_len; ++step) {
            scalar_t h_prev = (step == 0) ? hx[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] : layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx];
            
            // Compute gates
            scalar_t igate = 0, hgate = 0;
            int gate_offset = 0;
            
            for (int gate = 0; gate < 3; ++gate) {
                igate = 0;
                hgate = 0;
                
                // Input projection
                int input_dim = (layer == 0) ? input_size : hidden_size;
                for (int i = 0; i < input_dim; ++i) {
                    scalar_t x_val = (layer == 0) ? input[step * batch_size * input_size + batch_idx * input_size + i] : layer_hidden[(layer-1) * batch_size * hidden_size + batch_idx * hidden_size + i];
                    igate += x_val * layer_weights_ih[gate * hidden_size * input_dim + hid_idx * input_dim + i];
                }
                
                // Hidden projection
                for (int i = 0; i < hidden_size; ++i) {
                    scalar_t h_val = h_prev;
                    hgate += h_val * layer_weights_hh[gate * hidden_size * hidden_size + hid_idx * hidden_size + i];
                }
                
                // Add biases
                if (layer_bias_ih) igate += layer_bias_ih[gate * hidden_size + hid_idx];
                if (layer_bias_hh) hgate += layer_bias_hh[gate * hidden_size + hid_idx];
                
                scalar_t combined = igate + hgate;
                
                if (gate == 0) { // Reset gate
                    scalar_t r = sigmoid(combined);
                    layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = r;
                } else if (gate == 1) { // Update gate
                    scalar_t z = sigmoid(combined);
                    layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = z;
                } else { // New gate
                    scalar_t r = layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx];
                    scalar_t n = tanh(combined + r * layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx + hidden_size]);
                    scalar_t z = layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx + hidden_size];
                    scalar_t h_new = (1 - z) * n + z * h_prev;
                    
                    if (layer == num_layers - 1) {
                        output[step * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = h_new;
                    }
                    
                    final_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = h_new;
                }
            }
        }
    }
}
"""

gru_cpp_code = """
torch::Tensor fused_gru_forward(
    torch::Tensor input,
    torch::Tensor hx,
    std::vector<torch::Tensor> weights_ih,
    std::vector<torch::Tensor> weights_hh,
    std::vector<torch::Tensor> biases_ih,
    std::vector<torch::Tensor> biases_hh) {
    
    auto output = torch::zeros({input.size(0), input.size(1), hx.size(2)}, input.options());
    auto final_hidden = torch::zeros_like(hx);
    
    const int batch_size = input.size(1);
    const int hidden_size = hx.size(2);
    const int num_layers = hx.size(0);
    const int seq_len = input.size(0);
    const int input_size = input.size(2);
    
    // Prepare device arrays for weights/biases
    std::vector<const float*> weights_ih_ptrs;
    std::vector<const float*> weights_hh_ptrs;
    std::vector<const float*> biases_ih_ptrs;
    std::vector<const float*> biases_hh_ptrs;
    
    for (int i = 0; i < num_layers; ++i) {
        weights_ih_ptrs.push_back(weights_ih[i].data_ptr<float>());
        weights_hh_ptrs.push_back(weights_hh[i].data_ptr<float>());
        biases_ih_ptrs.push_back(biases_ih[i].defined() ? biases_ih[i].data_ptr<float>() : nullptr);
        biases_hh_ptrs.push_back(biases_hh[i].defined() ? biases_hh[i].data_ptr<float>() : nullptr);
    }
    
    // Copy pointers to device
    auto d_weights_ih = torch::from_blob((void*)weights_ih_ptrs.data(), {num_layers}, torch::kLong).to(input.device());
    auto d_weights_hh = torch::from_blob((void*)weights_hh_ptrs.data(), {num_layers}, torch::kLong).to(input.device());
    auto d_biases_ih = torch::from_blob((void*)biases_ih_ptrs.data(), {num_layers}, torch::kLong).to(input.device());
    auto d_biases_hh = torch::from_blob((void*)biases_hh_ptrs.data(), {num_layers}, torch::kLong).to(input.device());
    
    const int threads = BLOCK_SIZE;
    const int blocks = (batch_size * hidden_size + threads - 1) / threads;
    
    fused_gru_kernel<float><<<blocks, threads, num_layers * batch_size * hidden_size * sizeof(float)>>>(
        input.data_ptr<float>(),
        hx.data_ptr<float>(),
        (const float**)d_weights_ih.data_ptr<int64_t>(),
        (const float**)d_weights_hh.data_ptr<int64_t>(),
        (const float**)d_biases_ih.data_ptr<int64_t>(),
        (const float**)d_biases_hh.data_ptr<int64_t>(),
        output.data_ptr<float>(),
        final_hidden.data_ptr<float>(),
        seq_len,
        batch_size,
        input_size,
        hidden_size,
        num_layers);
    
    return {output, final_hidden};
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("fused_gru_forward", &fused_gru_forward, "Fused multi-layer GRU forward");
}

    return 1.0 / (1.0 + exp(-x));
}

__device__ scalar_t tanh(scalar_t x) {
    return ::tanh(x);
}

template <typename scalar_t>
__global__ void fused_gru_kernel(
    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ hx,
    const scalar_t** __restrict__ weights_ih,
    const scalar_t** __restrict__ weights_hh,
    const scalar_t** __restrict__ biases_ih,
    const scalar_t** __restrict__ biases_hh,
    scalar_t* __restrict__ output,
    scalar_t* __restrict__ final_hidden,
    int seq_len,
    int batch_size,
    int input_size,

    if (bid >= batch_size * hidden_size) return;

    const int hid_idx = bid % hidden_size;
    
    extern __shared__ char shared_mem[];

        const scalar_t* layer_weights_hh = weights_hh[layer];

        const scalar_t* layer_bias_hh = biases_hh ? biases_hh[layer] : nullptr;

            scalar_t igate, hgate;
            for (int gate = 0; gate < 3; ++gate) {

                hgate = 0;

                for (int i = 0; i < input_dim; ++i) {

#include <cuda.h>

template <typename scalar_t>
__device__ scalar_t sigmoid(scalar_t x) {
    return 1.0 / (1.0 + exp(-x));
}

template <typename scalar_t>
__device__ scalar_t tanh(scalar_t x) {
    return ::tanh(x);
}

template <typename scalar_t>
__global__ void fused_gru_kernel(
    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ hx,
    const scalar_t** __restrict__ weights_ih,
    const scalar_t** __restrict__ weights_hh,
    const scalar_t** __restrict__ biases_ih,
    const scalar_t** __restrict__ biases_hh,
    scalar_t* __restrict__ output,
    scalar_t* __restrict__ final_hidden,
    scalar_t* __restrict__ layer_hidden,
    int seq_len,
    int batch_size,
    int input_size,
    int hidden_size,
    int num_layers) {
    
    const int bid = blockIdx.x * blockDim.x + threadIdx.x;
    if (bid >= batch_size * hidden_size) return;
    
    const int batch_idx = bid / hidden_size;
    const int hid_idx = bid % hidden_size;
    
    for (int layer = 0; layer < num_layers; ++layer) {
        const scalar_t* layer_weights_ih = weights_ih[layer];
        const scalar_t* layer_weights_hh = weights_hh[layer];
        const scalar_t* layer_bias_ih = biases_ih ? biases_ih[layer] : nullptr;
        const scalar_t* layer_bias_hh = biases_hh ? biases_hh[layer] : nullptr;
        
        for (int step = 0; step < seq_len; ++step) {
            scalar_t h_prev = (step == 0) ? hx[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] 
                : layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx];
            
            scalar_t r, z, n;
            for (int gate = 0; gate < 3; ++gate) {
                scalar_t igate = 0, hgate = 0;
                
                int input_dim = (layer == 0) ? input_size : hidden_size;
                for (int i = 0; i < input_dim; ++i) {
                    scalar_t x_val = (layer == 0) ? input[step * batch_size * input_size + batch_idx * input_size + i] 
                        : layer_hidden[(layer-1) * batch_size * hidden_size + batch_idx * hidden_size + i];
                    igate += x_val * layer_weights_ih[gate * hidden_size * input_dim + hid_idx * input_dim + i];
                }
                
                for (int i = 0; i < hidden_size; ++i) {
                    scalar_t h_val = h_prev;
                    hgate += h_val * layer_weights_hh[gate * hidden_size * hidden_size + hid_idx * hidden_size + i];
                }
                
                if (layer_bias_ih) igate += layer_bias_ih[gate * hidden_size + hid_idx];
                if (layer_bias_hh) hgate += layer_bias_hh[gate * hidden_size + hid_idx];

                } else if (gate == 1) { 
                    z = sigmoid(igate + hgate);
                } else { 
                    n = tanh(igate + hgate * r);
                }
            }
            
            scalar_t h_new = (1 - z) * n + z * h_prev;
            layer_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = h_new;
            
            if (layer == num_layers - 1) {
                output[step * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = h_new;
            }
            final_hidden[layer * batch_size * hidden_size + batch_idx * hidden_size + hid_idx] = h_new;
        }
    }

}

__device__ scalar_t sigmoid(scalar_t x) {
    return 1.0 / (1.0 + exp(-x));
}

template <typename scalar_t>
__device__ scalar_t tanh(scalar_t x) {
    return ::tanh(x);
}

template <typename scalar_t>
__global__ void fused_gru_kernel(
    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ hx,
    const scalar_t** __restrict__ weights_ih,
    const scalar_t** __restrict__ weights_hh,
    const scalar_t** __restrict__ biases_ih,
    const scalar_t** __restrict__ biases_hh,
    scalar_t* __restrict__ output,
    scalar_t* __restrict__ final_hidden,
    scalar_t* __restrict__ layer_hidden,
    int seq_len,
    int batch_size,
    int input_size,
    int hidden_size,
    int num_layers) {
    
    const int bid = blockIdx.x * blockDim.x + threadIdx.x;
    if (bid >= batch_size * hidden_size) return;
    
    const int batch_idx = bid / hidden_size;
    const int hid_idx = bid % hidden_size;
    
    for (int layer = 0; layer < num_layers; ++layer) {
        const scalar_t* layer_weights_ih = weights_ih[layer];
        const scalar_t* layer_weights_hh = weights_hh[layer];
        const scalar_t* layer_bias_ih = biases_ih ? biases_ih[layer] : nullptr;
        const scalar_t* layer_bias_hh = biases_hh ? biases_hh[layer] : nullptr;

            scalar_t r, z, n;
            
            scalar_t igate = 0;

            for (int i = 0; i < input_dim; ++i) {

                igate += x_val * layer_weights_ih[0 * hidden_size * input_dim + hid_idx * input_dim + i];
            }
            if (layer_bias_ih) igate += layer_bias_ih[0 * hidden_size + hid_idx];

                igate += x_val * layer_weights_ih[1 * hidden_size * input_dim + hid_idx * input_dim + i];

__global__ void inclusive_scan_kernel(const float* input, float* output, int dim_size, int stride) {
    extern __shared__ float shared_buffers[];
    float* scan_array = &shared_buffers[0];
    float* temp_array = &shared_buffers[dim_size];
    
    int batch_idx = blockIdx.x;
    int tid = threadIdx.x;
    int offset = batch_idx * stride;
    
    // Load data into shared memory
    for (int i = tid; i < dim_size; i += blockDim.x) {

    // Hillis-Steele inclusive scan with double buffering
    for (int d = 1; d < dim_size; d *= 2) {
        for (int i = tid; i < dim_size; i += blockDim.x) {
            float val = scan_array[i];
            if (i >= d) {
                val += scan_array[i - d];
            }
            temp_array[i] = val;
        }
        __syncthreads();
        
        // Swap buffers
        float* swap = scan_array;
        scan_array = temp_array;
        temp_array = swap;

__device__ __forceinline__ float4 load_vector(const float* ptr) {
    return *reinterpret_cast<const float4*>(ptr);
}

__device__ __forceinline__ void store_vector(float* ptr, const float4& val) {
    *reinterpret_cast<float4*>(ptr) = val;
}

__global__ void inclusive_scan_kernel(const float* input, float* output, int dim_size, int stride) {
    extern __shared__ float shared_data[];
    const int PADDED_DIM = BLOCK_SIZE * ELEMENTS_PER_THREAD;
    float* warp_sums = &shared_data[PADDED_DIM];
    
    int batch_idx = blockIdx.x;
    int tid = threadIdx.x;
    int warp_id = tid / 32;
    int lane_id = tid % 32;
    int offset = batch_idx * stride;

    // Vectorized load with padding
    float4* shared_vec = reinterpret_cast<float4*>(shared_data);
    int vec_per_thread = ELEMENTS_PER_THREAD / VEC_SIZE;
    int total_vec = PADDED_DIM / VEC_SIZE;
    
    for(int i = 0; i < vec_per_thread; ++i) {
        int vec_idx = tid * vec_per_thread + i;
        if(vec_idx < total_vec) {
            int global_idx = offset + vec_idx * VEC_SIZE;
            float4 val = (global_idx < offset + dim_size) ? 
                        load_vector(input + global_idx) : make_float4(0,0,0,0);
            shared_vec[vec_idx] = val;
        }
    }
    __syncthreads();

    // Warp-level scan with shuffle instructions
    #pragma unroll
    for(int i = 0; i < ELEMENTS_PER_THREAD; ++i) {
        int local_idx = warp_id * 32 * ELEMENTS_PER_THREAD + lane_id * ELEMENTS_PER_THREAD + i;
        float val = shared_data[local_idx];
        
        for(int delta = 1; delta < 32; delta *= 2) {

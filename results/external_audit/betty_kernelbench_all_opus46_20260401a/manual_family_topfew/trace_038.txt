    int padding_h,
    int padding_w,
    int dilation_d,
    int dilation_h,
    int dilation_w
) {
    int index = blockIdx.x * blockDim.x + threadIdx.x;
    if (index >= batch_size * channels * output_d * output_h * output_w) return;

    // Unravel index
    int batch = index / (channels * output_d * output_h * output_w);
    int remainder = index % (channels * output_d * output_h * output_w);
    int ch = remainder / (output_d * output_h * output_w);
    remainder = remainder % (output_d * output_h * output_w);
    int d = remainder / (output_h * output_w);
    remainder = remainder % (output_h * output_w);
    int h = remainder / output_w;
    int w = remainder % output_w;

    int start_d = d * stride_d - padding_d;
    int start_h = h * stride_h - padding_h;
    int start_w = w * stride_w - padding_w;

    float max_val = -INFINITY;

    for (int kd = 0; kd < kernel_d; ++kd) {
        int id = start_d + kd * dilation_d;

) {
    const int index = blockIdx.x * blockDim.x + threadIdx.x;
    const int total_elements = batch_size * channels * output_d * output_h * output_w;
    if (index >= total_elements) return;

    // Optimized memory access pattern for coalescing
    int w = index % output_w;
    int residual = index / output_w;
    int h = residual % output_h;
    residual /= output_h;
    int d = residual % output_d;
    residual /= output_d;
    int ch = residual % channels;
    int batch = residual / channels;

    const int start_h = h * stride_h - padding_h;
    const int start_w = w * stride_w - padding_w;

    float max_val = -INFINITY;
    const int input_offset = (batch * channels + ch) * input_d * input_h * input_w;

    const int ch_volume = output_volume * channels;

    const int ch = (index % ch_volume) / output_volume;
    const int residual = index % output_volume;

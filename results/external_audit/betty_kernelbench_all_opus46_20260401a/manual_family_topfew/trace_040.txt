#include <cuda_runtime.h>

__global__ void avg_pool2d_kernel(
    const float* input,
    float* output,
    int batch_size,
    int channels,
    int input_height,
    int input_width,
    int output_height,
    int output_width,
    int kernel_size,
    int stride,
    int padding,
    float inv_kernel_area
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total_elements = batch_size * channels * output_height * output_width;
    if (idx >= total_elements) return;

    int c = idx % channels;
    int b = (idx / channels) / (output_height * output_width);
    int oh = (idx / channels) % output_height;
    int ow = (idx % (output_height * output_width)) % output_width;

    int h_start = oh * stride - padding;
    int w_start = ow * stride - padding;
    int h_end = h_start + kernel_size;
    int w_end = w_start + kernel_size;

    float sum = 0.0f;
    for (int h = h_start; h < h_end; ++h) {
        for (int w = w_start; w < w_end; ++w) {
            if (h >= 0 && h < input_height && w >= 0 && w < input_width) {
                int input_idx = ((b * channels + c) * input_height + h) * input_width + w;
                sum += input[input_idx];
            }
        }
    }
    output[idx] = sum * inv_kernel_area;
}

#include <cuda_runtime.h>

__global__ void avg_pool2d_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    int batch_channels,
    int input_height,

    int output_height,

) {
    const int output_elements = output_height * output_width;

#include <cuda_runtime.h>

__global__ void avg_pool2d_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    int batch_channels,
    int input_height,
    int input_width,
    int output_height,
    int output_width,
    int kernel_size,
    int stride,

) {
    const int output_elements = output_height * output_width;
    const int total_elements = batch_channels * output_elements;

        const int bc = idx / output_elements;
        const int output_idx = idx % output_elements;

        const int ow = output_idx % output_width;
        
        const int h_start = oh * stride - padding;

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cooperative_groups.h>
#include <cuda_fp16.h>

namespace cg = cooperative_groups;

__global__ void fused_matmul_dropout_reduce_softmax_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weights,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int in_features,
    int out_features,
    int batch_size,
    float dropout_prob,
    bool training,
    unsigned int seed) {

    const int batch_idx = blockIdx.x;
    const int tid = threadIdx.x;
    const int feature_stride = blockDim.x;

    extern __shared__ float shared_mem[];
    float* shared = shared_mem;

    float val = 0.0f;

    // Vectorized matmul with loop unrolling
    const int vec_size = 4;
    int remaining = in_features % vec_size;
    int vec_end = in_features - remaining;

    for(int i = tid; i < vec_end; i += feature_stride) {
        float4 in_vec = *reinterpret_cast<const float4*>(&input[batch_idx*in_features + i]);
        float4 w_vec = *reinterpret_cast<const float4*>(&weights[tid*in_features + i]);
        val += in_vec.x * w_vec.x + in_vec.y * w_vec.y + in_vec.z * w_vec.z + in_vec.w * w_vec.w;
    }

    // Handle remaining elements
    for(int i = vec_end + tid; i < in_features; i += feature_stride) {
        val += input[batch_idx*in_features + i] * weights[tid*in_features + i];
    }

    // Block-wide reduction for matmul results
    cg::thread_block tb = cg::this_thread_block();
    val = cg::reduce(tb, val, cg::plus<float>()) + bias[tid];

    if (training) {
        // Optimized dropout with XORWOW RNG
        unsigned int rng_state = (seed + batch_idx * 10007) ^ (tid * 12345);
        rng_state ^= rng_state << 13;
        rng_state ^= rng_state >> 17;
        rng_state ^= rng_state << 5;
        float rand_val = __uint_as_float(rng_state | 0x3f800000) - 1.0f;
        
        if (rand_val < dropout_prob) val = 0.0f;
        else val /= (1.0f - dropout_prob);
    }

    // Warp-level reduction for mean
    cg::thread_group warp = cg::tiled_partition<32>(tb);
    float warp_sum = cg::reduce(warp, val, cg::plus<float>());
    
    if (warp.thread_rank() == 0) {
        int warp_id = tid / 32;
        shared[warp_id] = warp_sum;

    tb.sync();

    if (tid == 0) {
        float total = 0.0f;
        int num_warps = (feature_stride + 31) / 32;
        for(int i = 0; i < num_warps; ++i) total += shared[i];
        output[batch_idx] = 1.0f;  // Softmax trivial for single element
    }
}

__global__ void fused_matmul_dropout_reduce_softmax_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weights,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int in_features,
    int out_features,
    int batch_size,
    float dropout_prob,
    bool training,
    unsigned int seed) {

    const int batch_idx = blockIdx.x;
    const int tid = threadIdx.x;
    const int feature_stride = blockDim.x;

    extern __shared__ float shared_mem[];
    float* shared = shared_mem;

    if (tid >= out_features) return;

    float val = 0.0f;

    // Vectorized matmul for aligned elements
    const int vec_size = 4;
    int remaining = in_features % vec_size;
    int vec_end = in_features - remaining;

    for(int i = tid; i < vec_end; i += feature_stride) {
        float4 in_vec = *reinterpret_cast<const float4*>(&input[batch_idx*in_features + i]);
        float4 w_vec = *reinterpret_cast<const float4*>(&weights[tid*in_features + i]);
        val += in_vec.x * w_vec.x + in_vec.y * w_vec.y + in_vec.z * w_vec.z + in_vec.w * w_vec.w;
    }

    // Handle remaining elements
    for(int i = vec_end + tid; i < in_features; i += feature_stride) {
        val += input[batch_idx*in_features + i] * weights[tid*in_features + i];
    }

    // Block-wide reduction using shared memory
    __shared__ float block_sum[1024];
    block_sum[tid] = val;
    __syncthreads();

    for(int s = feature_stride/2; s > 0; s >>= 1) {
        if(tid < s) {
            block_sum[tid] += block_sum[tid + s];
        }
        __syncthreads();
    }

    val = block_sum[0] + bias[tid];

    if (training) {
        // Philox-based random number generation
        unsigned long long rand_seed = (seed + batch_idx) * 6364136223846793005ULL + 1442695040888963407ULL;

    // Warp-level reduction for mean

namespace cg = cooperative_groups;

__global__ void fused_matmul_dropout_reduce_softmax_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weights,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int in_features,
    int out_features,
    int batch_size,
    float dropout_prob,
    bool training,
    unsigned int seed) {

    const int batch_idx = blockIdx.x;
    const int tid = threadIdx.x;
    
    if (tid >= out_features) return;

    float val = bias[tid];
    
    for(int i = tid; i < in_features; i += blockDim.x) {
        val += input[batch_idx*in_features + i] * weights[i*out_features + tid];
    }

    if (training) {
        unsigned int rng_state = (seed + batch_idx * 100 + tid) * 1664525u + 1013904223u;
        rng_state ^= (rng_state >> 16);
        float rand = __uint_as_float(0x3f800000 | (rng_state & 0x7fffff)) - 1.0f;
        if (rand < dropout_prob) val = 0.0f;
        else val *= 1.0f / (1.0f - dropout_prob);
    }

    cg::thread_block_tile<32> warp = cg::tiled_partition<32>(cg::this_thread_block());
    float warp_sum = cg::reduce(warp, val, cg::plus<float>());

    if (warp.thread_rank() == 0) {
        atomicAdd(&output[batch_idx], warp_sum / out_features);
    }
}
